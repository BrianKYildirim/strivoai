    ```typescript
    import { handleOptionsRequest, corsHeaders } from '../_shared/cors.ts';
    import { requireAuth } from '../_shared/auth.ts';
    import { getServiceClient } from '../_shared/supabase-client.ts';
    import { createLogger } from '../_shared/logger.ts';
    import { getStripeClient } from '../_shared/stripe-client.ts';

    const log = createLogger('create-stripe-portal');

    export async function handleCreateStripePortal(req: Request): Promise<Response> {
      if (req.method === 'OPTIONS') return handleOptionsRequest({ type: 'user' });
      if (req.method !== 'POST') {
        return new Response(JSON.stringify({ error: 'method_not_allowed' }), {
          status: 405,
          headers: { ...corsHeaders({ type: 'user' }), 'Content-Type': 'application/json' },
        });
      }

      let ctx;
      try { ctx = await requireAuth(req, { type: 'user' }); }
      catch (r) { return r as Response; }
      if (!ctx) return new Response(JSON.stringify({ error: 'Unauthorized' }), { status: 401 });

      const supabase = getServiceClient();
      const { data: profile } = await supabase
        .from('profiles').select('stripe_customer_id').eq('id', ctx.userId).single();

      const customerId = profile?.stripe_customer_id as string | null;
      if (!customerId) {
        return new Response(JSON.stringify({ error: 'no_customer' }), {
          status: 400,
          headers: { ...corsHeaders({ type: 'user' }), 'Content-Type': 'application/json' },
        });
      }

      const appUrl = (Deno.env.get('APP_URL') ?? '').replace(/\/$/, '');
      const stripe = getStripeClient();
      const session = await stripe.billingPortal.sessions.create({
        customer: customerId,
        return_url: `${appUrl}/billing`,
      });

      log.info('portal-session-created', { userId: ctx.userId, sessionId: session.id });

      return new Response(JSON.stringify({ url: session.url }), {
        status: 200,
        headers: { ...corsHeaders({ type: 'user' }), 'Content-Type': 'application/json' },
      });
    }

    Deno.serve(handleCreateStripePortal);
    ```

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-07-03 | Elevation of Privilege | unauthenticated portal access | mitigate | `requireAuth(req, { type: 'user' })` rejects missing/invalid JWT with 401. Profile lookup uses `ctx.userId` only — never a query parameter — so cross-user access is structurally impossible. |
| T-07-06 | Spoofing | cross-origin portal calls | mitigate | `handleOptionsRequest({ type: 'user' })` + `corsHeaders({ type: 'user' })` use the `APP_URL` allowlist from `_shared/cors.ts` (fail-closed). |
| T-07-07 | Information Disclosure | error responses / logs | mitigate | `400 no_customer` is the only error code surfaced; `STRIPE_SECRET_KEY` never logged; `createLogger` uses reason codes. |
| Stale `stripe_customer_id` → portal for deleted customer | Operational | accept | Stripe returns a clear error; handler surfaces 500 with generic message. Dashboard-side customer deletion is rare and a v2 concern. |
</threat_model>
