function isV3Grade(grading_data: unknown): boolean {
  if (!grading_data || typeof grading_data !== 'object') return false;
  const g = grading_data as Record<string, unknown>;
  return typeof g.rawScore === 'number' && Array.isArray(g.sectionBreakdown);
}

```typescript
// Inside checkAccessAndLoadCache, after auth + tier gating passes:
const { data: { user } } = await supabase.auth.getUser();
if (!user) { setEntitlementState('error'); return; }

// 1. Get billing cycle start via SQL RPC
const { data: billingStartData, error: billingErr } = await supabase.rpc('get_billing_period_start', {
  p_user_id: user.id,
});

if (billingErr) {
  console.error('[LinkedInGrader] billing-cycle query failed', billingErr);
  setEntitlementState('will-use-credit'); // graceful degrade per Pitfall 5
  // still attempt cache load below
}

const billingStart = billingStartData as string | null;

// 2. Single query for current-cycle rows, ordered desc
let cycleRows: Array<{ id: string; overall_score: number; grading_data: unknown; created_at: string; profile_url?: string | null }> = [];
if (billingStart) {
  const { data, error } = await supabase
    .from('linkedin_grades')
    .select('id, overall_score, grading_data, created_at, profile_url')
    .eq('user_id', user.id)
    .gte('created_at', billingStart)
    .order('created_at', { ascending: false });

  if (error) {
    console.error('[LinkedInGrader] cycle-count query failed', error);
    setEntitlementState('will-use-credit');
  } else {
    cycleRows = data ?? [];
    setEntitlementState(cycleRows.length === 0 ? 'free-available' : 'will-use-credit');
  }
} else {
  setEntitlementState('will-use-credit');
}

// 3. Cache-load: use row[0] if it's v3.0-shaped, otherwise skip cache (D-03)
const candidate = cycleRows[0];
if (candidate && isV3Grade(candidate.grading_data)) {
  const cached = candidate.grading_data as LinkedInResult;
  cached.recordId = candidate.id;
  cached.created_at = candidate.created_at;
  setResult(cached);
  // Any existing cache-loaded side effects (setProfileUrl, etc.) preserved
} else {
  // Legacy row or no row — leave input form visible per D-03
  setResult(null);
}
```
