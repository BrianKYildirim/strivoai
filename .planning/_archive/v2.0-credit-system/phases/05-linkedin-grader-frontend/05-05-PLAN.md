```typescript
import { getServiceClient } from './supabaseClient.ts';
import { createLogger } from './logger.ts';

const log = createLogger('billing-period');

/**
 * Wraps the `get_billing_period_start(p_user_id)` Postgres RPC.
 * Returns the ISO date string on success, null on error.
 *
 * Callers (e.g., grade-linkedin free-scan branch) must treat `null` as
 * "unknown billing start → fall through to paid path" (safer default).
 * Does NOT throw — errors are logged and swallowed.
 */
export async function getBillingPeriodStart(userId: string): Promise<string | null> {
  const supabase = getServiceClient();
  const { data, error } = await supabase.rpc('get_billing_period_start', { p_user_id: userId });

  if (error) {
    log.error('billing-period-rpc-failed', { userId, error: String(error) });
    return null;
  }

  if (typeof data !== 'string') {
    log.warn('billing-period-unexpected-shape', { userId, dataType: typeof data });
    return null;
  }

  return data;
}
```
