```typescript
async function invokeWithRetry(params: AnthropicInvokeParams, log: ReturnType<typeof createLogger>): Promise<AnthropicInvokeResult> {
  const delays = [1000, 2000, 3000];
  let lastResult!: AnthropicInvokeResult;
  for (let attempt = 0; attempt < 3; attempt++) {
    lastResult = await invokeAnthropicText(params);
    if (lastResult.ok) return lastResult;
    // Don't retry client errors (4xx except 429/529)
    if (!lastResult.ok && lastResult.status >= 400 && lastResult.status < 500 && lastResult.status !== 429) {
      return lastResult;
    }
    log.warn('ai-retry', { attempt: attempt + 1, status: lastResult.ok ? 0 : lastResult.status });
    if (attempt < 2) {
      await new Promise(r => setTimeout(r, delays[attempt]));
    }
  }
  return lastResult;
}
```
