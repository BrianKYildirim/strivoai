Exact replacement code for the init `useEffect` (replaces lines ~102-135):

```typescript
  // ── Data fetch on mount ────────────────────────────────────────
  useEffect(() => {
    const init = async () => {
      const { data: { user } } = await supabase.auth.getUser();
      if (!user) { navigate('/auth'); return; }

      // Parallel: profile + primary resume (independent)
      const [profileResult, primaryResumeResult] = await Promise.all([
        supabase.from('profiles')
          .select('subscription_tier')
          .eq('id', user.id).single(),
        supabase.from('resumes')
          .select('file_name, grading_data')
          .eq('user_id', user.id)
          .not('grading_data', 'is', null)
          .order('analyzed_at', { ascending: false })
          .limit(1)
          .maybeSingle(),
      ]);

      setTier(getEffectiveTier(profileResult.data?.subscription_tier || 'free'));

      let rawResp: Record<string, any> | null = null;
      let filename: string | null = null;

      if (primaryResumeResult.data?.grading_data) {
        rawResp = primaryResumeResult.data.grading_data as Record<string, any>;
        filename = primaryResumeResult.data.file_name ?? null;
      } else {
        // Fallback: legacy MinimalHero users (analysis_results only)
        const { data: fallbackAnalysis } = await supabase
          .from('analysis_results')
          .select('raw_response, source_reference')
          .eq('user_id', user.id)
          .eq('analysis_type', 'resume_analysis')
          .order('created_at', { ascending: false })
          .limit(1)
          .maybeSingle();

        if (fallbackAnalysis?.raw_response && typeof fallbackAnalysis.raw_response === 'object') {
          rawResp = fallbackAnalysis.raw_response as Record<string, any>;
          console.log('[RecruitingLens] Using analysis_results fallback path');

          // Filename lookup via source_reference (resume_id) with defense-in-depth user_id check
          if (fallbackAnalysis.source_reference) {
            const { data: resumeRow } = await supabase
              .from('resumes')
              .select('file_name')
              .eq('id', fallbackAnalysis.source_reference)
              .eq('user_id', user.id)
              .maybeSingle();
            filename = resumeRow?.file_name ?? null;
          }
        }
      }

      const hasAnalysis = !!rawResp;
      setHasAnalyzedResume(hasAnalysis);
      setResumeFilename(filename);
      setResumeSummary(hasAnalysis ? buildResumeSummary(rawResp!) : '');
      setLoading(false);
    };
    init();
  }, []);
```

Pitfalls to avoid (from RESEARCH.md 346-366):
1. Do NOT drop `.not('grading_data', 'is', null)` — a not-yet-analyzed upload would otherwise "win" the ORDER BY.
2. Guard `if (fallbackAnalysis.source_reference)` before the filename lookup — runtime may be null even if TS says otherwise.
3. Use `.maybeSingle()` on every query (NOT `.single()`).
4. Keep `.eq('user_id', user.id)` on the fallback filename lookup for defense in depth even though RLS covers it.
5. Do NOT extract a helper hook — single call site, CLAUDE.md says avoid unnecessary abstractions.
6. Do NOT modify `buildResumeSummary`, the right-column card JSX (lines ~390-440), or the `recruiting-lens` edge function — locked by D-09, D-08, D-10.
