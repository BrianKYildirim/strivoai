1. Create `supabase/functions/interview-prep-coach/validation.ts` exporting a pure function:
   ```typescript
   export interface ValidationResult {
     valid: boolean;
     error?: string;
   }

   const EXPECTED_DISTRIBUTION: Record<string, number> = {
     behavioral: 3,
     technical: 2,
     situational: 1,
     culture_fit: 1,
     leadership: 1,
   };

   const REQUIRED_FIELDS = [
     'id', 'category', 'difficulty', 'question', 'whyAsked',
     'starOutline', 'sampleOpener', 'tips', 'redFlags',
   ];

   const STAR_FIELDS = ['situation', 'task', 'action', 'result'];

   export function validateInterviewQuestions(parsed: unknown): ValidationResult {
     // Check top-level structure
     if (!parsed || typeof parsed !== 'object' || !Array.isArray((parsed as any).questions)) {
       return { valid: false, error: 'Response missing questions array' };
     }
     const questions = (parsed as any).questions;
     // Check count
     if (questions.length !== 8) {
       return { valid: false, error: `Expected 8 questions, got ${questions.length}` };
     }
     // Check category distribution
     const counts: Record<string, number> = {};
     for (const q of questions) {
       const cat = q.category;
       counts[cat] = (counts[cat] || 0) + 1;
     }
     for (const [cat, expected] of Object.entries(EXPECTED_DISTRIBUTION)) {
       if ((counts[cat] || 0) !== expected) {
         return { valid: false, error: `Expected ${expected} ${cat} questions, got ${counts[cat] || 0}` };
       }
     }
     // Check required fields per question
     for (const q of questions) {
       for (const field of REQUIRED_FIELDS) {
         if (!(field in q) || q[field] === null || q[field] === undefined) {
           return { valid: false, error: `Question ${q.id ?? '?'} missing field: ${field}` };
         }
       }
       // Check starOutline sub-fields
       if (typeof q.starOutline === 'object' && q.starOutline !== null) {
         for (const sf of STAR_FIELDS) {
           if (!(sf in q.starOutline)) {
             return { valid: false, error: `Question ${q.id ?? '?'} starOutline missing: ${sf}` };
           }
         }
       }
       // Check tips and redFlags are arrays
       if (!Array.isArray(q.tips)) {
         return { valid: false, error: `Question ${q.id ?? '?'}: tips must be an array` };
       }
       if (!Array.isArray(q.redFlags)) {
         return { valid: false, error: `Question ${q.id ?? '?'}: redFlags must be an array` };
       }
     }
     return { valid: true };
   }
   ```
