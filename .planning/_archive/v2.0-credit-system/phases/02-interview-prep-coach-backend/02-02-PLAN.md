```typescript
// Parse response — strip optional markdown fences, then repair if truncated
function parseAiResponse(rawText: string): { questions: unknown[] } {
  let text = rawText.trim();
  if (text.startsWith('```json')) {
    text = text.replace(/^```json\s*/, '').replace(/```\s*$/, '');
  } else if (text.startsWith('```')) {
    text = text.replace(/^```\s*/, '').replace(/```\s*$/, '');
  }
  try {
    return JSON.parse(text);
  } catch {
    const repaired = repairTruncatedJSON(text);
    return JSON.parse(repaired);
  }
}

let parsed = parseAiResponse(ai.text);
let validation = validateInterviewQuestions(parsed);

if (!validation.valid) {
  log.warn('validation-failed-first-attempt', { error: validation.error });
  // Retry AI call once (D-12)
  const retryAi = await invokeAnthropicText({
    model: ANTHROPIC_MODEL,
    system: systemPrompt,
    messages: [{ role: 'user', content: userMessage }],
    maxTokens: 8000,
    temperature: 0.7,
    fixtureKey: 'interview_prep',
  });

  if (!retryAi.ok) {
    log.error('retry-ai-failed', { status: retryAi.status, error: retryAi.error });
    return new Response(
      JSON.stringify({ error: 'Failed to generate valid interview questions after retry.' }),
      { status: 500, headers: { ...corsHeaders({ type: 'user' }), 'Content-Type': 'application/json' } },
    );
  }

  parsed = parseAiResponse(retryAi.text);
  validation = validateInterviewQuestions(parsed);

  if (!validation.valid) {
    log.error('validation-failed-second-attempt', { error: validation.error });
    return new Response(
      JSON.stringify({ error: 'Failed to generate valid interview questions. Please try again.' }),
      { status: 500, headers: { ...corsHeaders({ type: 'user' }), 'Content-Type': 'application/json' } },
    );
  }
}

log.info('request-complete', { questionCount: parsed.questions?.length });

return new Response(
  JSON.stringify(parsed),
  { headers: { ...corsHeaders({ type: 'user' }), 'Content-Type': 'application/json' } },
);
```
