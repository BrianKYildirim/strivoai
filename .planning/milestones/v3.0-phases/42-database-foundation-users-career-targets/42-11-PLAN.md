<threat_model>
- **T-42-BENCHMARKS** — Small-cohort de-anonymization risk. Mitigation: rollup function applies WHERE sample_size>=5 BEFORE upsert; benchmarks.sample_size CHECK enforces at row level (defense in depth); aggregate-only columns (percentiles, sample_size) — no individual identifiers leak; documented in function COMMENT.
- **T-42-CRON** — Idempotency + GUC dependency. Mitigation: cron job invocation is pure SQL (no HTTP — no GUCs needed for THIS cron specifically, simpler than catchup/lifecycle crons); cron.unschedule guard re-schedules idempotently; SECURITY DEFINER function executes with elevated privileges (service-role-equivalent) but is scoped to read-only-then-upsert against benchmarks (no destructive ops on user data).
- **T-42-RLS** — SECURITY DEFINER function bypasses RLS by design. Mitigation: function body restricts to documented metric computation + benchmarks upsert; does NOT expose individual user_id values in outputs; function REVOKE EXECUTE from PUBLIC + GRANT to no one (only service role / pg_cron context invokes it).
</threat_model>

    (3) CREATE OR REPLACE FUNCTION public.run_benchmarks_rollup() RETURNS jsonb LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$ DECLARE v_cohort_count integer := 0; v_rows_upserted integer := 0; v_skipped integer := 0; BEGIN
      WITH 
        -- Step 1: build cohorts from cohort_peers
        cohorts AS (
          SELECT 
            lower(trim(coalesce(school,''))) || '|' || lower(trim(coalesce(year,''))) || '|' || lower(trim(coalesce(major,''))) || '|' || lower(trim(coalesce(t_role,''))) || '|' || lower(trim(coalesce(industry,''))) || '|' || lower(trim(coalesce(location_preference,''))) AS cohort_key,
            user_id, school, year, major, t_role, industry, location_preference
          FROM (
            SELECT cp.user_id, cp.school, cp.year, cp.major, cp.location_preference,
                   unnest(coalesce(cp.target_roles, '{NULL}'::text[])) AS t_role,
                   unnest(coalesce(cp.target_industries, '{NULL}'::text[])) AS industry
            FROM public.cohort_peers cp
          ) expanded
        ),
        -- Step 2: per-cohort score samples
        score_samples AS (
          SELECT c.cohort_key,
                 s.overall_score AS resume_overall_score,
                 s.ats_score,
                 s.quantified_impact_score
          FROM cohorts c
          JOIN public.resume_versions rv ON rv.user_id = c.user_id
          JOIN public.scores s ON s.version_id = rv.id AND s.version_type = 'resume'
        ),
        -- Step 3: per-cohort linkedin score samples
        linkedin_samples AS (
          SELECT c.cohort_key, s.overall_score AS linkedin_overall_score
          FROM cohorts c
          JOIN public.linkedin_versions lv ON lv.user_id = c.user_id
          JOIN public.scores s ON s.version_id = lv.id AND s.version_type = 'linkedin'
        ),
        -- Step 4: per-cohort outcome rates
        outcome_samples AS (
          SELECT c.cohort_key,
                 count(*) FILTER (WHERE o.result = 'interview') :: numeric / NULLIF(count(*),0) AS interview_rate,
                 count(*) FILTER (WHERE o.result = 'offer') :: numeric / NULLIF(count(*),0) AS offer_rate
          FROM cohorts c
          JOIN public.applications a ON a.user_id = c.user_id
          JOIN public.outcomes o ON o.application_id = a.id
          GROUP BY c.cohort_key
        ),
        -- Step 5: per-metric percentiles with sample_size filter
        metric_rollups AS (
          SELECT cohort_key, 'resume_overall_score' AS metric,
                 percentile_cont(0.25) WITHIN GROUP (ORDER BY resume_overall_score) AS p25,
                 percentile_cont(0.50) WITHIN GROUP (ORDER BY resume_overall_score) AS p50,
                 percentile_cont(0.75) WITHIN GROUP (ORDER BY resume_overall_score) AS p75,
                 count(*) AS sample_size
          FROM score_samples
          GROUP BY cohort_key
          UNION ALL
          SELECT cohort_key, 'ats_score', percentile_cont(0.25) WITHIN GROUP (ORDER BY ats_score), percentile_cont(0.50) WITHIN GROUP (ORDER BY ats_score), percentile_cont(0.75) WITHIN GROUP (ORDER BY ats_score), count(*) FROM score_samples GROUP BY cohort_key
          UNION ALL
          SELECT cohort_key, 'quantified_impact_score', percentile_cont(0.25) WITHIN GROUP (ORDER BY quantified_impact_score), percentile_cont(0.50) WITHIN GROUP (ORDER BY quantified_impact_score), percentile_cont(0.75) WITHIN GROUP (ORDER BY quantified_impact_score), count(*) FROM score_samples GROUP BY cohort_key
          UNION ALL
          SELECT cohort_key, 'linkedin_overall_score', percentile_cont(0.25) WITHIN GROUP (ORDER BY linkedin_overall_score), percentile_cont(0.50) WITHIN GROUP (ORDER BY linkedin_overall_score), percentile_cont(0.75) WITHIN GROUP (ORDER BY linkedin_overall_score), count(*) FROM linkedin_samples GROUP BY cohort_key
          UNION ALL
          SELECT cohort_key, 'interview_rate', interview_rate, interview_rate, interview_rate, (SELECT count(*) FROM cohorts c2 WHERE c2.cohort_key = o.cohort_key)
          FROM outcome_samples o
          UNION ALL
          SELECT cohort_key, 'offer_rate', offer_rate, offer_rate, offer_rate, (SELECT count(*) FROM cohorts c2 WHERE c2.cohort_key = o.cohort_key)
          FROM outcome_samples o
        )
      -- Step 6: UPSERT into benchmarks (sample_size>=5 filter is the DOCUMENTED EXCEPTION de-anon barrier)
      INSERT INTO public.benchmarks (cohort_key, metric, percentile_25, percentile_50, percentile_75, sample_size, computed_at)
      SELECT cohort_key, metric, p25, p50, p75, sample_size, now()
      FROM metric_rollups
      WHERE sample_size >= 5  -- T-42-BENCHMARKS de-anon barrier
      ON CONFLICT (cohort_key, metric, (computed_at::date))
      DO UPDATE SET percentile_25 = EXCLUDED.percentile_25, percentile_50 = EXCLUDED.percentile_50, percentile_75 = EXCLUDED.percentile_75, sample_size = EXCLUDED.sample_size;
      GET DIAGNOSTICS v_rows_upserted = ROW_COUNT;
      SELECT count(DISTINCT cohort_key) INTO v_cohort_count FROM cohorts;
      RAISE NOTICE 'run_benchmarks_rollup completed: cohort_count=%, rows_upserted=%, skipped_below_threshold=(filter-implicit)', v_cohort_count, v_rows_upserted;
      RETURN jsonb_build_object('cohort_count', v_cohort_count, 'rows_upserted', v_rows_upserted);
    END; $$;
    REVOKE ALL ON FUNCTION public.run_benchmarks_rollup() FROM PUBLIC;
    -- Function invoked by pg_cron (which runs as postgres role, not authenticated)
    
    (4) COMMENT ON FUNCTION public.run_benchmarks_rollup() IS 'Phase 42 L4 DB-BMK-02. Nightly cohort percentile rollup. cohort_key = lower-pipe-joined (school|year|major|target_role|industry|location). sample_size>=5 filter is DOCUMENTED EXCEPTION T-42-BENCHMARKS de-anonymization barrier. SECURITY DEFINER + REVOKE FROM PUBLIC limits invocation to pg_cron + service role.';
    
    (5) DO $cron$ BEGIN
      IF EXISTS (SELECT 1 FROM cron.job WHERE jobname = 'benchmarks_nightly') THEN
        PERFORM cron.unschedule('benchmarks_nightly');
      END IF;
      PERFORM cron.schedule(
        'benchmarks_nightly',
        '0 3 * * *',  -- 03:00 UTC nightly (Phase 12 + keep-warm slot alignment)
        $job$ SELECT public.run_benchmarks_rollup(); $job$
      );
    END $cron$;

RAISE EXCEPTION on mismatch.
