### Step 7 — `withMerits` wrapper (replace body inside callback)

```typescript
if (!MERIT_SYSTEM_ENABLED) {
  return jsonResponse({ error: "merit_disabled" }, 503);   // defensive — should never hit in prod
}

const gated = await withMerits(
  { userId, tier: userContext.tier as 'pro' | 'elite', corsType: 'user' as const },
  'interview_prep',
  async (_txId) => {
    const promptParams = {
      resumeText,
      jobDescription: sanitizedJD,
      backgroundContext: sanitizedBg,
      roleTitle: role_title,
      company,
      interviewType: interview_type,
      questionCategories: question_categories,
      difficulty,
      duration,
      format,
      variant,
    };
    const systemPrompt = loadPromptVariant('interview', userContext.tier as Tier, promptParams);

    const maxTokens = variant === 'pro' ? 5000 : 10000;   // D-06
    const model = getModel(userContext.tier as Tier, 'interview');   // Haiku 4.5
    const timeoutMs = getTimeout(userContext.tier as Tier, 'quiz');  // 80s

    // First attempt at temp 0.4
    let parsed: unknown;
    let validation = { valid: false, error: 'unstarted' } as { valid: boolean; error?: string; payload?: InterviewPrepPayload };
    const firstAi = await invokeAnthropicText({
      model, system: systemPrompt,
      messages: [{ role: 'user', content: 'Generate the interview prep content based on the instructions above.' }],
      maxTokens, temperature: 0.4,
      fixtureKey: 'interview_prep',
      timeoutMs,
    });
    if (!firstAi.ok) throw new MeritRefundable(firstAi.status, firstAi.error);
    try { parsed = parseAiResponse(firstAi.text); }
    catch { throw new MeritRefundable(502, 'JSON parse failed on first attempt'); }
    validation = validateInterviewPrepPayload(parsed, variant, intake.data);

    // Retry once at temp 0.2 if validation fails OR if no-silent-drops flags (D-11)
    let droppedFields: string[] = [];
    if (validation.valid && validation.payload) {
      droppedFields = findDroppedOptionalFields(validation.payload, {
        role_title, company, background_context,
      });
    }
    const needsRetry = !validation.valid || droppedFields.length > 0;
    if (needsRetry) {
      log.warn('interview-prep-retry', {
        reason: !validation.valid ? 'schema' : 'silent-drops',
        error: validation.error,
        droppedFields,
      });
      const retryAi = await invokeAnthropicText({
        model, system: systemPrompt,
        messages: [{ role: 'user', content: 'Generate the interview prep content. Return ONLY valid JSON with no extra text. Reference role_title / company / background_context explicitly in coach_verdict or focus_areas when they are provided.' }],
        maxTokens, temperature: 0.2,
        fixtureKey: 'interview_prep',
        timeoutMs,
      });
      if (!retryAi.ok) throw new MeritRefundable(retryAi.status, retryAi.error);
      try { parsed = parseAiResponse(retryAi.text); }
      catch { throw new MeritRefundable(502, 'JSON parse failed on retry'); }
      validation = validateInterviewPrepPayload(parsed, variant, intake.data);
      if (!validation.valid) throw new MeritRefundable(502, `validation_failed: ${validation.error}`);
      // Re-check silent drops; D-11: ship + warn at second miss (do NOT throw)
      if (validation.payload) {
        const secondDrops = findDroppedOptionalFields(validation.payload, { role_title, company, background_context });
        if (secondDrops.length > 0) {
          log.warn('silent-drop-detected', { droppedFields: secondDrops });
        }
      }
    }

    const payload = validation.payload!;   // safe: validation.valid === true at this point

    // Persistence: interview_quizzes + (variant-dependent) answers + streak
    const quizInsert = await supabase.from('interview_quizzes').insert({
      user_id: userId,
      mode: variant === 'pro' ? 'flashcard' : 'quiz',
      tier: variant,
      job_description,
      interview_type,
      question_categories,
      difficulty,
      duration,
      format,
      background_context: background_context ?? null,
      role_title: role_title ?? null,
      company: company ?? null,
      readiness_score: payload.your_prep_readiness.score,
    }).select('id').single();
    if (quizInsert.error || !quizInsert.data) {
      log.error('quiz-insert-failed', { error: quizInsert.error?.message });
      throw new MeritRefundable(500, 'quiz_persist_failed');
    }
    const quizId = quizInsert.data.id;

    // Questions: only for quiz variants (elite / elite_taste) per D-07
    if (payload.practice_questions.mode === 'quiz') {
      const rows = payload.practice_questions.items.map((q, i) => ({
        quiz_id: quizId,
        question_order: i + 1,             // SC-7: integer, matches evaluator parseInt join
        question_text: q.question,
        competency: q.competency,
      }));
      await supabase.from('interview_quiz_answers').insert(rows);
    }

    // Streak (only variant==='elite' advances streaks per Phase 19 D-12 carried forward)
    if (variant === 'elite') {
      const { data: existing } = await supabase.from('interview_streaks').select('*').eq('user_id', userId).maybeSingle();
      const nowIso = new Date().toISOString();
      const midnight = todayMidnightUtc();
      const last = existing?.last_quiz_at;
      const isFirstAfterMidnight = !last || new Date(last) < new Date(midnight);
      const hoursSinceLast = last ? (Date.now() - new Date(last).getTime()) / 3_600_000 : Infinity;
      const coldExpired = hoursSinceLast >= 48;
      const newStreak = coldExpired ? 1
        : (isFirstAfterMidnight ? (existing?.current_streak ?? 0) + 1 : (existing?.current_streak ?? 0));
      const newQuizzesToday = isFirstAfterMidnight ? 1 : (existing?.quizzes_today ?? 0) + 1;
      const newLongest = Math.max(existing?.longest_streak ?? 0, newStreak);
      await supabase.from('interview_streaks').upsert({
        user_id: userId,
        current_streak: newStreak,
        longest_streak: newLongest,
        quizzes_today: newQuizzesToday,
        last_quiz_at: nowIso,
        updated_at: nowIso,
      });
    }

    // record_ai_usage (Phase 22-04 Pitfall 2 fix — History visibility)
    await supabase.rpc('record_ai_usage', { p_user_id: userId, p_action_type: 'interview_prep', p_credits: 1 })
      .then(({ error }) => { if (error) log.error('usage-tracking-insert-failed', { error: String(error) }); });

    // Build response — compute todayQuizCount for Elite daily_usage.used (re-query since the write above increments the count)
    let todayQuizCount = 0;
    if (variant === 'elite') {
      const midnight = todayMidnightUtc();
      const { count } = await supabase
        .from('interview_quizzes')
        .select('id', { count: 'exact', head: true })
        .eq('user_id', userId)
        .eq('tier', 'elite')
        .gte('created_at', midnight);
      todayQuizCount = (count ?? 0) - 1;   // subtract the just-inserted row — daily_usage.used below re-adds via `+ 1`
    }
    const streakRow = variant === 'elite'
      ? (await supabase.from('interview_streaks').select('current_streak, longest_streak, quizzes_today, last_quiz_at').eq('user_id', userId).maybeSingle()).data
      : null;
    const response: {
      quiz_id: string;
      variant: typeof variant;
      payload: InterviewPrepPayload;
      streak: { current_streak: number; longest_streak: number; quizzes_today: number; daily_cap: number; last_quiz_at: string | null } | null;
      daily_usage: { used: number; cap: number } | null;
    } = {
      quiz_id: quizId,
      variant,
      payload,
      streak: streakRow ? {
        current_streak: streakRow.current_streak,
        longest_streak: streakRow.longest_streak,
        quizzes_today: streakRow.quizzes_today,
        daily_cap: 3,
        last_quiz_at: streakRow.last_quiz_at,
      } : null,
      daily_usage: variant === 'elite' ? { used: todayQuizCount + 1, cap: 3 } : null,
    };
    return jsonResponse(response, 200);
  },
);
if (!gated.ok) return gated.response;
return gated.value;
```
