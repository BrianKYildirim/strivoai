```typescript
const RUN = Deno.env.get('RUN_EVALS') === '1';
Deno.test({ ignore: !RUN, name: 'honest-scoring: Cohen kappa >= 0.7 on top 3 dimensions' }, async () => {
  const lines = (await Deno.readTextFile('.../weak-fit-30.jsonl')).trim().split('\n').map((l) => JSON.parse(l));
  const results: Array<{ id: string; labels: any; predicted: any }> = [];
  for (const ex of lines) {
    const payload = await invokeAnalyzeHandlerWithFixture(ex); // real Anthropic call; no fixture replay for eval
    results.push({ id: ex.id, labels: ex.labels, predicted: extractDimensions(payload) });
  }
  for (const dim of ['role_fit_score','experience_score','technical_score']) {
    const kappa = cohenKappa(
      results.map(r => bucket(r.labels[dim])),
      results.map(r => bucket(r.predicted[dim])),
    );
    assert(kappa >= 0.7, `${dim} kappa ${kappa} < 0.7`);
  }
});
```

`bucket(n)` maps score→category: `n<40→'low'`, `n<70→'mid'`, `n>=70→'high'`. `cohenKappa` is an inline 20-line implementation (Wikipedia formula; no external dep).

CI does NOT run this by default (Anthropic cost). It runs on-demand via `RUN_EVALS=1 deno test ...` and before major prompt changes to prevent honest-scoring regression.
