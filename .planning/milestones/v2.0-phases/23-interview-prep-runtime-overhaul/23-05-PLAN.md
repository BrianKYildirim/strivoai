function buildBasePrompt(params: InterviewPrepPromptParams, opts: {
  mode: 'flashcard' | 'quiz';
  questionCount: number;        // 3 for Pro-flash, duration→5/5/6/7 for quiz
  budgetHint: string;            // 'concise / 5K tokens' vs 'expanded / 10K tokens'
}): string {
  return `
You are Strivo's Interview Prep Coach. Output a Runtime §10 nine-section JSON payload for the candidate below.

=== CANDIDATE RESUME ===
${params.resumeText.slice(0, 8000)}

=== JOB DESCRIPTION ===
${params.jobDescription.slice(0, 6000)}

=== INTAKE ===
- role_title: ${params.roleTitle ?? '(not provided)'}
- company: ${params.company ?? '(not provided)'}
- interview_type: ${params.interviewType}
- question_categories: ${params.questionCategories.join(', ')}
- difficulty: ${params.difficulty}
- duration_minutes: ${params.duration}
- format: ${params.format}
- background_context: ${params.backgroundContext ?? '(not provided)'}

=== OUTPUT CONTRACT (Runtime §10) ===
Return ONLY a JSON object with these nine top-level keys, in this order:
1. your_prep_readiness: { score: 0-100, band: "not_ready"|"needs_work"|"ready_to_practice"|"ready_to_ship", one_line_verdict: string (20-240 chars) }
2. coach_verdict: string (120-1200 chars) — honest plain-text assessment. If the candidate is weak-fit, say so; do NOT sugarcoat.
3. readiness_breakdown: { resume_alignment: 0-100, jd_alignment: 0-100, star_depth: 0-100, communication: 0-100, notes: string (40-800 chars) }
4. focus_areas: array of 2-5 { title, why, what_to_do } objects
5. quick_tips: array of 3-6 strings (10-240 chars each)
6. practice_questions: { mode: "${opts.mode}", items: [ ... ] } — see MODE SECTION below
7. prep_roadmap: array of 3-8 { when: "today"|"this_week"|"before_interview", action: string }
8. strengths_and_watchouts: { strengths: string[2-5], watch_outs: string[2-5] }
9. coaching_advice: { advice: string (60-800 chars), biggest_score_mover: string (20-300 chars) }

=== MODE SECTION ===
${opts.mode === 'flashcard'
  ? `practice_questions.items must be exactly 3 flashcards with shape:
{ question, competency, star_answer: { situation, task, action, result, resume_anchor, signal_missing }, coach_note (<=140 chars) }
- Each STAR beat 40-1200 chars. If the resume does not support the beat with evidence, set signal_missing=true and resume_anchor=null.
- Runtime §5: every STAR beat MUST cite resume_anchor (short quote from the resume) OR set signal_missing=true — never both omitted.`
  : `practice_questions.items must contain exactly ${opts.questionCount} questions (duration=${params.duration} minutes per Runtime §7.2).
Each item: { id: integer 1..${opts.questionCount}, question: 10-500 chars, competency: 2-60 chars }.
Do NOT include answers — answers are produced by the user and graded separately.`}

=== PRINCIPLES (Runtime §11 / §5) ===
- Honest scoring: if the resume is weak-fit, the readiness band must be "not_ready" or "needs_work". Do not anchor to 75.
- Reference the provided intake fields in coach_verdict or focus_areas when they are filled. If role_title/company/background_context are provided, mention them explicitly at least once (D-11 no-silent-drops).
- Never fabricate resume content; cite it.
- Token budget: ${opts.budgetHint}.

Return ONLY the JSON object. No markdown fences, no commentary.`.trim();
}

function buildProFlashcardPrompt(p: InterviewPrepPromptParams): string {
  return buildBasePrompt(p, { mode: 'flashcard', questionCount: 3, budgetHint: 'concise, fits in 5K tokens' });
}

function durationToCount(d: 15 | 30 | 45 | 60): 5 | 5 | 6 | 7 {
  // Runtime §7.2: 15→5, 30→5, 45→6, 60→7
  if (d === 15 || d === 30) return 5;
  if (d === 45) return 6;
  return 7;
}

function buildEliteQuizPrompt(p: InterviewPrepPromptParams): string {
  return buildBasePrompt(p, {
    mode: 'quiz',
    questionCount: durationToCount(p.duration),
    budgetHint: 'expanded — up to 10K tokens; use the space for richer coach_verdict and prep_roadmap',
  });
}

function buildEliteTastePrompt(p: InterviewPrepPromptParams): string {
  // Same structure as Elite quiz; Pro tier gets a taste of Elite-grade output.
  // No demotion — content parity with Elite is the point of the taste branch.
  return buildEliteQuizPrompt(p);
}

export const INTERVIEW_PREP_VARIANTS: PromptVariantBuilder<InterviewPrepPromptParams> = {
  free: () => 'ERROR: Free tier should not call interview prep. This prompt should never reach Anthropic — the handler must 403 before loadPromptVariant.',
  pro: (params) => params.variant === 'elite_taste'
    ? buildEliteTastePrompt(params)
    : buildProFlashcardPrompt(params),
  elite: (params) => buildEliteQuizPrompt(params),
};

registerFeaturePrompts<InterviewPrepPromptParams>('interview', INTERVIEW_PREP_VARIANTS);
