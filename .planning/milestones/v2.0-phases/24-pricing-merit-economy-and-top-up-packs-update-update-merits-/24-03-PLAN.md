```typescript
// ── Phase 24 · D-03: FEATURE_KEY_TO_DB — bridges existing short FeatureKey
//    ('resume'/'career'/...) to the canonical SC-12 FeatureKey
//    ('resumeAnalysis'/'careerCompass'/...) that pricing-config.ts uses.
//    Use the canonical key when calling getMaxOutput / getMaxInputTokens / etc.

export const FEATURE_KEY_TO_DB: Record<FeatureKey, CanonicalFeatureKey> = {
  resume:     'resumeAnalysis',
  career:     'careerCompass',
  linkedin:   'linkedinGrader',
  recruiting: 'recruitingLens',
  interview:  'interviewPrepCoach',

// ── Phase 24 · D-03: verifyCostMap — called at module init against a snapshot
//    of `merit_feature_costs` rows; asserts DB ↔ TS parity. Pure function for
//    testability; the caller owns the DB read (done in Plan 04 wired at
//    cold-start in each edge-function bootstrap).

export interface MeritFeatureCostRow {
  feature: string;        // DB stores short names e.g. 'resume'
  tier: Tier;
  cost: number | null;
}

export interface VerifyCostMapResult {
  ok: boolean;
  drift?: Array<{ feature: string; tier: Tier; ts: number | null; db: number | null }>;
  missing?: Array<string>;  // 'feature/tier' strings
}

export function verifyCostMap(rows: MeritFeatureCostRow[]): VerifyCostMapResult {
  const drift: VerifyCostMapResult['drift'] = [];
  const missing: string[] = [];
  const tiers: Tier[] = ['free', 'pro', 'elite'];

  for (const shortKey of Object.keys(FEATURE_KEY_TO_DB) as FeatureKey[]) {
    const canonical = FEATURE_KEY_TO_DB[shortKey];
    const tsCosts = MERIT_COST_PER_FEATURE[canonical];
    for (const tier of tiers) {
      const row = rows.find((r) => r.feature === shortKey && r.tier === tier);
      if (!row) {
        missing.push(`${shortKey}/${tier}`);
        continue;
      }
      const tsVal = tsCosts[tier];
      if (row.cost !== tsVal) {
        drift.push({ feature: shortKey, tier, ts: tsVal, db: row.cost });
      }
    }
  }

  if (drift.length || missing.length) {
    return {
      ok: false,
      ...(drift.length ? { drift } : {}),
      ...(missing.length ? { missing } : {}),
    };
  }
  return { ok: true };
}
```

**Create `supabase/functions/_shared/input-guard.ts`**:

```typescript
/**
 * input-guard.ts — Phase 24 · D-12 · Single chokepoint every feature handler
 * MUST call before any Claude invocation.
 *
 * D-13: null cap on EITHER maxTokensInput or maxCharactersInput = tier-locked.
 * D-27: fail-closed — always resolve locked-tier state first, before any text work.
 * D-26: module-load only (no per-request import cost beyond the accessor call).
 */

import {
  getMaxInputChars,
  getMaxInputTokens,
  type CanonicalFeatureKey,
} from './tier-config.ts';
import type { Tier } from './tier-config.ts';

export type InputGuardResult =
  | { ok: true }
  | { ok: false; reason: 'feature-locked' }
  | { ok: false; reason: 'over-budget'; limit: number; actual: number };

export function validateInputBudget(
  feature: CanonicalFeatureKey,
  tier: Tier,
  text: string,
): InputGuardResult {
  // D-27 fail-closed: feature-locked resolves BEFORE any text inspection.
  const inputTokens = getMaxInputTokens(feature, tier);
  const inputChars = getMaxInputChars(feature, tier);
  if (inputTokens === null || inputChars === null) {
    return { ok: false, reason: 'feature-locked' };
  }

  // D-12 chokepoint: char-length is the authoritative budget (4:1 heuristic
  // per SC-12 invariant; tokens are the derived constraint).
  const actual = typeof text === 'string' ? text.length : 0;
  if (actual > inputChars) {
    return { ok: false, reason: 'over-budget', limit: inputChars, actual };
  }

  return { ok: true };
}

/**
 * Convenience helper — maps an InputGuardResult to a canonical edge-function
 * Response. Handlers can `if (!guard.ok) return toInputGuardResponse(guard);`.
 */
export function toInputGuardResponse(
  result: Exclude<InputGuardResult, { ok: true }>,
): Response {
  const { reason } = result;
  if (reason === 'feature-locked') {
    return new Response(
      JSON.stringify({
        error: 'feature-locked',
        message: 'This feature is not available on your current tier.',
      }),
      {
        status: 402,
        headers: { 'Content-Type': 'application/json' },
      },
    );
  }
  // over-budget
  return new Response(
    JSON.stringify({
      error: 'over-budget',
      message: `Input exceeds the maximum of ${result.limit} characters.`,
      limit: result.limit,
      actual: result.actual,
    }),
    {
      status: 413,
      headers: { 'Content-Type': 'application/json' },
    },
  );
}
```
