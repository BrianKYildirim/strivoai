    ```typescript
    import { describe, it, expect } from 'vitest';
    import { readFileSync, readdirSync } from 'node:fs';
    import { join } from 'node:path';

    const PLANNING_ROOT = join(process.cwd(), '.planning');
    const REQUIREMENTS_PATH = join(PLANNING_ROOT, 'REQUIREMENTS.md');
    const PHASE_DIR = join(PLANNING_ROOT, 'phases', '24-pricing-merit-economy-and-top-up-packs-update-update-merits-');

    function parseReqIdsFromRequirements(): string[] {
      const text = readFileSync(REQUIREMENTS_PATH, 'utf8');
      // Match table rows like `| TOPUP-01 | 24 | ... |`
      const phase24Rows = text.match(/\|\s*([A-Z]+-\d+)\s*\|\s*24\s*\|/g) ?? [];
      return phase24Rows
        .map(row => {
          const m = row.match(/\|\s*([A-Z]+-\d+)\s*\|/);
          return m ? m[1] : null;
        })
        .filter((x): x is string => x !== null);
    }

    function parseReqIdsFromPlanFrontmatter(planPath: string): string[] {
      const text = readFileSync(planPath, 'utf8');
      const fm = text.match(/^---\n([\s\S]*?)\n---/);
      if (!fm) return [];
      // Match `  - TOPUP-01` lines under the requirements: block
      const reqBlockMatch = fm[1].match(/requirements:\n((?:\s+-\s+[A-Z]+-\d+\n?)+)/);
      if (!reqBlockMatch) return [];
      return (reqBlockMatch[1].match(/[A-Z]+-\d+/g)) ?? [];
    }

    describe('Phase 24 REQ-ID coverage', () => {
      it('every REQ-ID in REQUIREMENTS.md with phase=24 appears in at least one plan', () => {
        const requiredIds = parseReqIdsFromRequirements();
        expect(requiredIds.length).toBeGreaterThan(0);

        const planFiles = readdirSync(PHASE_DIR)
          .filter(f => /^24-\d{2}-PLAN\.md$/.test(f))
          .map(f => join(PHASE_DIR, f));
        expect(planFiles.length).toBeGreaterThanOrEqual(7);

        const coveredIds = new Set<string>();
        for (const p of planFiles) {
          for (const id of parseReqIdsFromPlanFrontmatter(p)) coveredIds.add(id);
        }

        const orphans = requiredIds.filter(id => !coveredIds.has(id));
        expect(orphans, `Orphan REQ-IDs (in REQUIREMENTS.md but no plan closes them): ${orphans.join(', ')}`).toEqual([]);
      });

      it('every REQ-ID claimed by a plan exists in REQUIREMENTS.md', () => {
        const requiredIds = new Set(parseReqIdsFromRequirements());
        const planFiles = readdirSync(PHASE_DIR)
          .filter(f => /^24-\d{2}-PLAN\.md$/.test(f))
          .map(f => join(PHASE_DIR, f));

        const phantom: Array<{ plan: string; id: string }> = [];
        // Allowlist prior-phase REQ-IDs that Plan 03 LEGITIMATELY re-closes (BACK-08..11 from earlier phases)
        const priorPhaseAllowlist = new Set(['BACK-08', 'BACK-09', 'BACK-10', 'BACK-11', 'SHOP-09', 'SHOP-10', 'SHOP-11', 'MER-14', 'NFR-06', 'NFR-07', 'NFR-08', 'SUB-02', 'SUB-04', 'SHOP-01', 'SHOP-02', 'SHOP-03', 'SHOP-04', 'UX-08', 'UX-01', 'UX-04']);
        for (const p of planFiles) {
          for (const id of parseReqIdsFromPlanFrontmatter(p)) {
            if (!requiredIds.has(id) && !priorPhaseAllowlist.has(id)) {
              phantom.push({ plan: p.split(/[\\/]/).pop() ?? p, id });
            }
          }
        }
        expect(phantom, `Phantom REQ-IDs (claimed by plans but not in REQUIREMENTS.md): ${JSON.stringify(phantom)}`).toEqual([]);
      });
    });
    ```
