  shop_balance: number;
  ledgerRows: Array<{ balance_type: string; amount: number; deduction_id: string; source: string }>;

function makeFakeSupabase(state: FakeState) {
  return {
    rpc(fn: string, args: Record<string, unknown>) {
      if (fn === 'check_and_deduct_merits') {
        const cost = args.p_cost as number;
        const deductionId = (args.p_deduction_id as string) ?? crypto.randomUUID();
        if (state.subscription_balance + state.shop_balance < cost) {
          return Promise.resolve({ data: { error: 'insufficient_funds' }, error: null });
        }
        const fromSub = Math.min(cost, state.subscription_balance);
        const fromShop = cost - fromSub;
        state.subscription_balance -= fromSub;
        state.shop_balance -= fromShop;
        if (fromSub > 0) {
          state.ledgerRows.push({ balance_type: 'subscription', amount: -fromSub, deduction_id: deductionId, source: 'spend' });
        }
        if (fromShop > 0) {
          state.ledgerRows.push({ balance_type: 'shop', amount: -fromShop, deduction_id: deductionId, source: 'spend' });
        }
        return Promise.resolve({
          data: {
            success: true,
            subscription_spent: fromSub,
            shop_spent: fromShop,
            deduction_id: deductionId,
          },
          error: null,
        });
      }
      return Promise.resolve({ data: null, error: null });
    },
  };
}

import { describe, it, expect } from 'vitest';
import {
  verifyCostMap,
  FEATURE_KEY_TO_DB,
  type MeritFeatureCostRow,
} from '../tier-config';
import { MERIT_COST_PER_FEATURE } from '@/config/pricing-config';

/**
 * Simulates the DB snapshot that a feature handler would SELECT at cold-start.
 * In production the edge function does `supabase.from('merit_feature_costs').select(...)`
 * and feeds the result into verifyCostMap; failure throws at module-init.
 */
function dbSnapshot(overrides?: Partial<Record<string, number | null>>, omit?: Set<string>): MeritFeatureCostRow[] {
  const rows: MeritFeatureCostRow[] = [];
  for (const short of Object.keys(FEATURE_KEY_TO_DB) as Array<keyof typeof FEATURE_KEY_TO_DB>) {
    const canonical = FEATURE_KEY_TO_DB[short];
    const costs = MERIT_COST_PER_FEATURE[canonical];
    for (const tier of ['free', 'pro', 'elite'] as const) {
      const key = `${short}/${tier}`;
      if (omit?.has(key)) continue;
      const baseCost = costs[tier];
      const cost = overrides && key in overrides ? (overrides as Record<string, number | null>)[key] : baseCost;
      rows.push({ feature: short, tier, cost });
    }
  }
  return rows;
}

describe('D-03 verifyCostMap integration (simulated cold-start DB snapshot)', () => {
  it('matching snapshot -> ok:true', () => {
    expect(verifyCostMap(dbSnapshot())).toEqual({ ok: true });

  tier: string;
  cost: number | null;
}

// Short → canonical bridge (mirrors FEATURE_KEY_TO_DB on the edge side).
const SHORT_TO_CANONICAL: Record<string, CanonicalFeatureKey> = {
  resume:     'resumeAnalysis',
  career:     'careerCompass',
  linkedin:   'linkedinGrader',
  recruiting: 'recruitingLens',
  interview:  'interviewPrepCoach',
};

function tsModuleFallback(): FeatureCost[] {
  const rows: FeatureCost[] = [];
  for (const [short, canonical] of Object.entries(SHORT_TO_CANONICAL)) {
    const costs = MERIT_COST_PER_FEATURE[canonical];
    for (const tier of ['free', 'pro', 'elite'] as const) {
      rows.push({ feature: short, tier, cost: costs[tier] });
    }
  }
  return rows;
}

export function useFeatureCosts() {
  return useQuery({
    queryKey: ['feature-costs'],
    queryFn: async () => {
      const { data, error } = await supabase
        .from('merit_feature_costs')
        .select('feature, tier, cost');
      if (error || !data || data.length === 0) {
        // D-03 / ECON-02: TS module is authoritative; DB is the mirror.
        // A transient DB failure MUST NOT break the UI — use the TS fallback.
        return tsModuleFallback();
      }
      return data as FeatureCost[];
    },
    staleTime: 60_000,
    // Always give consumers something to render even before first fetch.
    initialData: tsModuleFallback,
  });
}

export function getCostForFeature(
  costs: FeatureCost[] | undefined,
  feature: string,
  tier: string,
): number | null {
  if (!costs) {
    // Last-line fallback — consumers that pass undefined still get a sane answer.
    const canonical = SHORT_TO_CANONICAL[feature];
    if (!canonical) return null;
    return MERIT_COST_PER_FEATURE[canonical][tier as Tier] ?? null;
  }
  const entry = costs.find((c) => c.feature === feature && c.tier === tier);
  return entry?.cost ?? null;
}
