### 2.1 · `src/components/pricing/TierCardV2.tsx`

```typescript
import { Button } from '@/components/ui/button';
import { PRICING, MERITS_PER_MONTH, MERIT_COST_PER_FEATURE, type Tier } from '@/config/pricing-config';
import { cn } from '@/lib/utils';

interface TierCardV2Props {
  tier: Tier;
  interval: 'monthly' | 'annual';
  isCurrent?: boolean;
  onCta: (tier: Tier, interval: 'monthly' | 'annual') => void;
}

const TIER_COPY: Record<Tier, { name: string; tagline: string; bullets: string[]; cta: string }> = {
  free: {
    name: 'Free',
    tagline: 'Try the platform',
    bullets: [
      'Resume analysis, Career Compass, LinkedIn Grader',
      `${MERITS_PER_MONTH.free} merits/month`,
      'Interview Prep + Recruiting Lens locked',
    ],
    cta: 'Start free',
  },
  pro: {
    name: 'Pro',
    tagline: 'Serious job seekers',
    bullets: [
      'All 5 AI features unlocked',
      `${MERITS_PER_MONTH.pro} merits/month`,
      'Priority support',
    ],
    cta: 'Upgrade to Pro',
  },
  elite: {
    name: 'Elite',
    tagline: 'Power users + executives',
    bullets: [
      'All Pro features, deeper token budgets',
      `${MERITS_PER_MONTH.elite} merits/month`,
      'Premium support + Interview Prep Quiz',
    ],
    cta: 'Upgrade to Elite',
  },
};

function fmtUsd(cents: number): string {
  return `$${(cents / 100).toFixed(2)}`;
}

export function TierCardV2({ tier, interval, isCurrent = false, onCta }: TierCardV2Props) {
  const copy = TIER_COPY[tier];
  const price = PRICING[tier][interval];
  const priceLabel = tier === 'free' ? 'Free forever' : `${fmtUsd(price)}/${interval === 'monthly' ? 'mo' : 'yr'}`;

  return (
    <article
      className={cn(
        'flex flex-col p-8 bg-[--surface-card] border border-[--surface-elevated] rounded-[--radius-boxy]',
        'transition-transform',
        tier === 'pro' && 'scale-[1.02] border-[--accent-red]', // spec-locked highlight on Pro
      )}
      aria-label={`${copy.name} tier card`}
    >
      <header className="mb-6">
        <h3 className="font-display text-2xl font-semibold">{copy.name}</h3>
        <p className="text-[--text-muted] mt-1">{copy.tagline}</p>
      </header>

      <div className="mb-6">
        <span className="font-mono text-3xl">{priceLabel}</span>
      </div>

      <ul className="flex-1 space-y-2 mb-8">
        {copy.bullets.map((b, i) => (
          <li key={i} className="text-sm text-[--text-primary]">• {b}</li>
        ))}
      </ul>

      <Button
        variant={tier === 'free' ? 'boxy-outline' : 'boxy'}
        onClick={() => onCta(tier, interval)}
        disabled={isCurrent}
        aria-label={`${copy.cta}${isCurrent ? ' (current plan)' : ''}`}
      >
        {isCurrent ? 'Current plan' : copy.cta}
      </Button>
    </article>
  );
}
```

import { describe, it, expect } from 'vitest';
import { readFileSync } from 'node:fs';
import { globSync } from 'node:fs';

const BUTTON_PATH = 'src/components/ui/button.tsx';
const PRICING_SURFACES = [
  'src/pages/Pricing.tsx',
  ...['TierCardV2', 'BillingIntervalToggleV2', 'PackSectionV2', 'PackCardV2'].map(
    (name) => `src/components/pricing/${name}.tsx`,
  ),
];

describe('D-18 boxy-lock lint', () => {
  it('button.tsx exposes boxy + boxy-outline variants', () => {
    const src = readFileSync(BUTTON_PATH, 'utf8');
    expect(src).toMatch(/\bboxy:\s*"/);
    expect(src).toMatch(/"boxy-outline":\s*"/);
  });

  it('no /pricing surface file uses rounded-full in className', () => {
    const offenders: string[] = [];
    for (const path of PRICING_SURFACES) {
      const src = readFileSync(path, 'utf8');
      if (/\brounded-full\b/.test(src)) offenders.push(path);
    }
    expect(offenders, `rounded-full forbidden on /pricing surfaces; offenders: ${offenders.join(', ')}`).toEqual([]);
  });

  it('no /pricing surface file uses variant="default" on Button', () => {
    const offenders: string[] = [];
    for (const path of PRICING_SURFACES) {
      const src = readFileSync(path, 'utf8');
      if (/variant=["']default["']/.test(src)) offenders.push(path);
    }
    expect(offenders, `variant="default" forbidden on /pricing surfaces; offenders: ${offenders.join(', ')}`).toEqual([]);
  });

  it('every <Button ... variant=...> under src/components/pricing/ uses boxy or boxy-outline', () => {
    const re = /<Button[^>]*variant=["']([^"']+)["']/g;
    const bad: Array<{ file: string; variant: string }> = [];
    for (const path of PRICING_SURFACES) {
      const src = readFileSync(path, 'utf8');
      let m: RegExpExecArray | null;
      while ((m = re.exec(src))) {
        if (m[1] !== 'boxy' && m[1] !== 'boxy-outline') {
          bad.push({ file: path, variant: m[1] });
        }
      }
    }
    expect(bad, `non-boxy variants found: ${JSON.stringify(bad)}`).toEqual([]);
  });

  it('every /pricing surface button has an aria-label (a11y sanity)', () => {
    // Cheap heuristic: if the file imports Button and renders <Button, at least one aria-label exists.
    for (const path of PRICING_SURFACES) {
      const src = readFileSync(path, 'utf8');
      if (/<Button/.test(src)) {
        expect(src, `${path} has <Button but no aria-label`).toMatch(/aria-label=/);
      }
    }
  });
});
