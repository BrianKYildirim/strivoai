    **Body of `handleCreatePaymentIntent`** — same shape as `handleCreateCheckoutSession` but simpler (one shape, no subscription branch). Key divergence: the body is validated against `PACKS_BY_ID` (not env price IDs) because top-up packs derive their price from the pricing-config.ts source of truth (D-01, D-02).

    ```typescript
    function jsonResponse(body: unknown, status: number): Response {
      return new Response(JSON.stringify(body), {
        status,
        headers: {
          ...corsHeaders({ type: "user" }),
          "Content-Type": "application/json",
        },
      });
    }

    export async function handleCreatePaymentIntent(
      req: Request,
      deps: CreatePaymentIntentDeps = {},
    ): Promise<Response> {
      if (req.method === "OPTIONS") {
        return handleOptionsRequest({ type: "user" });
      }
      if (req.method !== "POST") {
        return jsonResponse({ error: "method_not_allowed" }, 405);
      }

      // Auth — propagate Response on failure
      const requireAuthFn = deps.requireAuth ??
        ((r: Request) =>
          realRequireAuth(r, { type: "user" }) as Promise<UserContext>);
      let ctx: UserContext;
      try {
        ctx = await requireAuthFn(req);
      } catch (r) {
        if (r instanceof Response) return r;
        throw r;
      }

      // Rate limit — shop-tight cap 10/hr all tiers (matches SHOP-12 pattern)
      const rateLimitFn = deps.checkRateLimit ?? realCheckRateLimit;
      const rl = await rateLimitFn(ctx.userId, ctx.tier, {
        key: "create-payment-intent",
        limits: { free: 10, pro: 10, elite: 10 },
        windowSeconds: 3600,
      });
      if (!rl.allowed) return createRateLimitResponse(rl.retryAfterSeconds, "user");

      // Body
      const rawBody = await req.json().catch(() => null);
      const body = normalizeBody(rawBody);
      if (!body) return jsonResponse({ error: "invalid_body" }, 400);

      const pack = PACKS_BY_ID[body.pack_id];
      if (!pack) return jsonResponse({ error: "invalid_pack_id" }, 400);

      const supabase = deps.supabase ?? getServiceClient();
      const stripe = deps.stripe ?? getStripeClient();

      // Lazy customer creation (D-21 — copied from create-checkout-session lines 198-219)
      const { data: profile } = await supabase
        .from("profiles")
        .select("stripe_customer_id, email")
        .eq("id", ctx.userId)
        .single();

      let customerId: string | null =
        (profile?.stripe_customer_id as string | null) ?? null;

      if (!customerId) {
        const customer = await stripe.customers.create({
          email: ctx.email,
          metadata: { user_id: ctx.userId },
        });
        customerId = customer.id;
        await supabase.from("profiles")
          .update({
            stripe_customer_id: customerId,
            updated_at: new Date().toISOString(),
          })
          .eq("id", ctx.userId);
      }

      // Create PaymentIntent — metadata carries user_id + pack_id + pack_size so
      // the Plan 02 webhook branch on payment_intent.succeeded can credit Merits
      // idempotently via credit_shop_merits(p_user_id, p_pack_size, p_stripe_payment_id).
      let pi: Stripe.PaymentIntent;
      try {
        pi = await stripe.paymentIntents.create({
          amount: pack.priceCents,
          currency: "usd",
          customer: customerId!,
          automatic_payment_methods: { enabled: true },
          metadata: {
            user_id: ctx.userId,
            pack_id: pack.id,
            pack_size: String(pack.merits),
          },
        });
      } catch (err) {
        log.error("payment-intent-creation-failed", {
          userId: ctx.userId,
          packId: pack.id,
          err: err instanceof Error ? err.message : String(err),
        });
        return jsonResponse({ error: "payment_intent_creation_failed" }, 500);
      }

      log.info("payment-intent-created", {
        userId: ctx.userId,
        packId: pack.id,
        packSize: pack.merits,
        priceCents: pack.priceCents,
        paymentIntentId: pi.id,
      });

      return jsonResponse({
        clientSecret: pi.client_secret,
        paymentIntentId: pi.id,
      }, 200);
    }

    Deno.serve((req) => handleCreatePaymentIntent(req));
    ```

    function PaymentForm({ packId, onSuccess }: { packId: PackId; onSuccess: () => void }) {
      const stripe = useStripe();
      const elements = useElements();
      const [submitting, setSubmitting] = useState(false);
      const queryClient = useQueryClient();
      const pack = PACKS_BY_ID[packId];

        e.preventDefault();

          redirect: 'if_required',
        });

          return;

          queryClient.invalidateQueries({ queryKey: ['merit-balance'] });
          onSuccess();
        } else if (paymentIntent?.status === 'processing') {
          toast.info('Processing… your merits will appear shortly.');
          queryClient.invalidateQueries({ queryKey: ['merit-balance'] });
          onSuccess();
        }
        setSubmitting(false);
      };

      return (

          <PaymentElement />
          <Button
            type="submit"
            variant="boxy"
            disabled={!stripe || !elements || submitting}
            className="w-full"
          >

    import { useCallback, useEffect, useState } from 'react';
    import { PACKS_BY_ID, type PackId } from '@/config/pricing-config';
    import { useCreatePaymentIntent } from '@/hooks/useCreatePaymentIntent';
    import { StripePaymentElementModal } from '@/components/topup/StripePaymentElementModal';

    function isPackId(v: unknown): v is PackId {
      return typeof v === 'string' && v in PACKS_BY_ID;
    }

    export function TopUpProvider({ children }: { children: React.ReactNode }) {
      const [open, setOpen] = useState(false);
      const [packId, setPackId] = useState<PackId | null>(null);
      const [clientSecret, setClientSecret] = useState<string | null>(null);
      const { create } = useCreatePaymentIntent();

      const handleOpenTopUp = useCallback(async (event: Event) => {
        const detail = (event as CustomEvent<{ packId: unknown }>).detail;
        if (!detail || !isPackId(detail.packId)) {
          console.warn('[TopUpProvider] invalid packId in topup:open event', detail);
          return;
        }
        setPackId(detail.packId);
        setClientSecret(null);
        setOpen(true);
        const result = await create(detail.packId);
        if (!result) {
          setOpen(false);
          setPackId(null);
          return;
        }
        setClientSecret(result.clientSecret);
      }, [create]);

      useEffect(() => {
        window.addEventListener('topup:open', handleOpenTopUp);
        return () => window.removeEventListener('topup:open', handleOpenTopUp);
      }, [handleOpenTopUp]);

      const handleOpenChange = useCallback((next: boolean) => {
        setOpen(next);
        if (!next) {
          setPackId(null);
          setClientSecret(null);
        }
      }, []);

      return (
        <>
          {children}
          <StripePaymentElementModal
            open={open}
            onOpenChange={handleOpenChange}
            packId={packId}
            clientSecret={clientSecret}
          />
        </>
      );
    }

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser → supabase.functions.invoke('create-payment-intent') | Untrusted body `{ pack_id }` arrives over HTTPS with Bearer JWT. |
| Browser → Stripe PaymentElement (confirmPayment) | Client calls Stripe directly with clientSecret; Stripe authenticates via the PaymentIntent ownership. |
| Stripe → stripe-webhook-handler | Signed webhook event arrives; Plan 02 layers idempotency. This plan does NOT modify the webhook — it relies on Plan 02. |
| TopUpProvider (window event listener) → StripePaymentElementModal | In-page CustomEvent dispatch from PackCardV2 (Plan 05). Detail is untrusted because a curious browser console could craft arbitrary events. |
| usePaywallGuard → OutOfMeritsModalV2 | Server-side 402 response body `{ cost, balance, feature }` passes through to the modal. Feature name is interpolated into user-visible copy. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-24-25 | Tampering | Client-supplied `pack_id` in create-payment-intent body | mitigate | Server validates against `PACKS_BY_ID` via `isPackId()` guard — unknown ID returns 400 `invalid_pack_id`. Price is derived server-side from `PACKS_BY_ID[pack_id].priceCents` (D-01/D-02), not from the request. Even if a client sends `pack_id: 'pack_25'` with a forged price, the server uses the canonical price. |
| T-24-26 | Spoofing | Client-supplied `user_id` in PaymentIntent metadata | mitigate | Handler sets `metadata.user_id = ctx.userId` from the authenticated JWT — the body shape does not accept user_id at all. `normalizeBody` strips unknown fields. Stripe rejects metadata mutations after confirmation. |
| T-24-27 | Repudiation | User disputes a top-up purchase ("I never bought this") | mitigate | Three-layer audit trail: (a) `payment-intent-created` structured log at edge-function time with userId + packId + paymentIntentId; (b) Stripe PaymentIntent record is authoritative; (c) Plan 02's `merit_transactions` row with `source='topup'` + `stripe_payment_intent_id` + full ledger. Any auditor can cross-reference Stripe dashboard ↔ `merit_transactions`. |
| T-24-28 | Information Disclosure | PaymentIntent error surfaces stack traces or Stripe internals to client | mitigate | On `paymentIntents.create` failure, handler logs `err.message` server-side but returns bare `{ error: 'payment_intent_creation_failed' }` to client (fail-closed per D-27). `log.error` only in edge logs, never in the response body. |
| T-24-29 | Denial of Service | Attacker spams create-payment-intent to burn Stripe quota or generate phantom PaymentIntents | mitigate | Shop-tight rate limit: `{ free: 10, pro: 10, elite: 10 }` per hour (matches create-checkout-session SHOP-12 pattern). Additional: requireAuth gate makes unauthenticated spam impossible. If hit, returns 429 with Retry-After. |
| T-24-30 | Elevation of Privilege | Malicious browser console fires `window.dispatchEvent(new CustomEvent('topup:open', { detail: { packId: 'pack_25', __bypass: true } }))` attempting to manipulate the modal | mitigate | TopUpProvider's `isPackId()` guard rejects unknown shapes. Even with a valid packId injection, the attack produces nothing beyond what the PackCardV2 Buy button does — the user still must complete Stripe PaymentElement confirmation with a valid card. No authorization escalation possible. |
| T-24-31 | Tampering | Attacker intercepts and modifies the `clientSecret` in transit | mitigate | HTTPS-only (Supabase edge functions enforce TLS). `clientSecret` is single-use per PaymentIntent and bound to a specific amount/currency on Stripe's side. A tampered secret fails the Stripe SDK's handshake with an opaque error. |
| T-24-32 | Information Disclosure | `OutOfMeritsModalV2` body echoes `feature` name from 402 response — XSS vector if feature name contains HTML? | mitigate | React renders `{feature}` as text (JSX escapes by default). The 402 response body comes from our own edge functions with enum-bounded feature names (`'Resume Analysis'`, `'Career Compass'`, etc. — NOT user-supplied). Defence-in-depth: if a future backend bug introduces user content into `feature`, React still escapes. |
| T-24-33 | Repudiation | User claims "I never saw the Out-of-Merits modal" | accept | Modal dismissal is not audited client-side (no event ping). Acceptable because the 402 response itself is logged server-side — the authoritative event is the feature function refusing to run, not the modal render. Adding client telemetry is out-of-scope for Phase 24. |
| T-24-34 | Integrity | Stripe PaymentIntent succeeds but webhook delivery fails (never credits shop_balance) | transfer | Transferred to Plan 02's webhook retry + idempotency layers. If webhook is lost entirely, Stripe auto-retries for 3 days (exponential backoff). Operational monitoring on `merit_shop_purchases` row count vs. Stripe dashboard top-up count is the audit surface. This plan's frontend shows "Processing…" toast when confirm resolves but balance query doesn't update within 2s, giving the user an honest status. |
