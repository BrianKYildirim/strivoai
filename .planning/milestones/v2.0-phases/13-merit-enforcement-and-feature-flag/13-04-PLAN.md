-- Phase 13: Merit enforcement observability
-- Adds: 4 aggregation views, alert_state debounce table, 2 helper RPCs
-- Non-destructive; no data migration.

begin;

-- ─── View 1: Merits in circulation ───────────────────────────────────────────
-- Tier-bucketed sum of net impulses (deducted minus refunded) over trailing 30d
create or replace view public.v_merits_in_circulation as
select
  coalesce(p.tier, 'free')::text as tier,
  count(distinct mt.user_id)      as user_count,
  coalesce(sum(case
    when mt.status = 'deducted' then mt.amount_impulses
    when mt.status = 'refunded' then -mt.amount_impulses
    else 0 end), 0)               as net_impulses_30d
from public.merit_transactions mt
left join public.profiles p on p.id = mt.user_id
where mt.created_at >= now() - interval '30 days'
group by coalesce(p.tier, 'free');

-- ─── View 2: Feature consumption daily ───────────────────────────────────────
create or replace view public.v_feature_consumption_daily as
select
  date_trunc('day', mt.created_at)::date as day,
  mt.feature::text                       as feature,
  count(*) filter (where mt.status = 'deducted')  as deductions,
  count(*) filter (where mt.status = 'refunded')  as refunds,
  coalesce(sum(mt.amount_impulses) filter (where mt.status = 'deducted'), 0) as impulses_spent
from public.merit_transactions mt
where mt.created_at >= now() - interval '30 days'
group by 1, 2
order by 1 desc, 2;

-- ─── View 3: Shop purchases daily ────────────────────────────────────────────
-- PROVISIONAL: depends on Phase 12 shop_purchases table. If that table does NOT
-- yet exist, SKIP this view (comment the block) and add a follow-up gap plan.
create or replace view public.v_shop_purchases_daily as
select
  date_trunc('day', sp.created_at)::date as day,
  sp.currency::text                      as currency,
  count(*)                               as purchase_count,
  coalesce(sum(sp.amount), 0)            as total_amount,
  coalesce(sum(sp.impulses_granted), 0)  as total_impulses_granted
from public.shop_purchases sp
where sp.created_at >= now() - interval '30 days'
group by 1, 2
order by 1 desc, 2;

-- ─── View 4: Conversion funnel (PROVISIONAL per O-1) ─────────────────────────
-- Depends on Phase 12 merit_exhaustion_events. If absent, ship this view as a
-- stub returning zero rows (do NOT fail the migration).
create or replace view public.v_conversion_funnel as
select
  date_trunc('day', e.occurred_at)::date as day,
  count(distinct e.user_id) filter (where e.event_type = 'depleted')      as depleted_users,
  count(distinct e.user_id) filter (where e.event_type = 'shop_view')     as shop_viewers,
  count(distinct e.user_id) filter (where e.event_type = 'shop_purchase') as purchasers
from public.merit_exhaustion_events e
where e.occurred_at >= now() - interval '30 days'
group by 1
order by 1 desc;

-- ─── alert_state debounce table ──────────────────────────────────────────────
create table if not exists public.alert_state (
  alert_key    text primary key,
  last_fired_at timestamptz not null default now(),
  last_value   jsonb
);

alter table public.alert_state enable row level security;
-- No policies: service role only (matches Phase 10 helper-table pattern).

-- ─── Helper RPC: AI failure rate over trailing hour ──────────────────────────
create or replace function public.get_ai_failure_rate_1h()
returns numeric
language sql stable security definer
set search_path = public
as $$
  select case
    when count(*) = 0 then 0::numeric
    else (
      count(*) filter (where status = 'refunded')::numeric
      / nullif(count(*) filter (where status in ('deducted','refunded')), 0)
    )
  end
  from public.merit_transactions
  where created_at >= now() - interval '1 hour'
    and feature in (
      'resume_analysis','career_compass','linkedin_grading',
      'recruiting_lens','skills_gap_analysis','job_comparison','interview_prep'
    );
$$;

revoke all on function public.get_ai_failure_rate_1h() from public;
grant execute on function public.get_ai_failure_rate_1h() to service_role;

-- ─── Helper RPC: Shop purchase failure rate over trailing hour ───────────────
-- PROVISIONAL per O-1: guard body on shop_purchases_failed table existence.
create or replace function public.get_shop_failure_rate_1h()
returns numeric
language plpgsql stable security definer
set search_path = public
as $$
declare
  failed_tbl_exists boolean;
  success_count int;
  failed_count int;
begin
  select exists (
    select 1 from pg_tables where schemaname='public' and tablename='shop_purchases_failed'
  ) into failed_tbl_exists;

  if not failed_tbl_exists then
    return 0::numeric;
  end if;

  execute 'select count(*) from public.shop_purchases where created_at >= now() - interval ''1 hour'''
    into success_count;
  execute 'select count(*) from public.shop_purchases_failed where created_at >= now() - interval ''1 hour'''
    into failed_count;

  if (success_count + failed_count) = 0 then
    return 0::numeric;
  end if;
  return failed_count::numeric / (success_count + failed_count);
end;
$$;

revoke all on function public.get_shop_failure_rate_1h() from public;
grant execute on function public.get_shop_failure_rate_1h() to service_role;

-- ─── Analytics index (only if not already present from Phase 10) ─────────────
create index if not exists idx_merit_transactions_created_at
  on public.merit_transactions (created_at desc);

commit;

```ts
import { handleRequest } from '../_shared/handler.ts';
import { getServiceClient } from '../_shared/supabaseClient.ts';
import { createLogger } from '../_shared/logger.ts';

const log = createLogger('admin-merit-metrics');

async function isAdmin(userId: string, email: string | null): Promise<boolean> {
  const envList = (Deno.env.get('ADMIN_EMAILS') ?? '').split(',').map(s => s.trim().toLowerCase()).filter(Boolean);
  if (email && envList.includes(email.toLowerCase())) return true;
  // Fallback: user_roles table
  const svc = getServiceClient();
  const { data, error } = await svc.from('user_roles').select('role').eq('user_id', userId).eq('role', 'admin').maybeSingle();
  if (error) {
    log.warn('admin-check-fallback-error', { userId, error: error.message });
    return false;
  }
  return !!data;
}

export default handleRequest({
  authMode: 'user',
  corsType: 'internal',
  handler: async ({ userContext, req }) => {
    if (!userContext) return new Response(JSON.stringify({ error: 'unauthorized' }), { status: 401 });
    // Pull email from JWT claim (handleRequest should populate this; otherwise look up via service client)
    const svc = getServiceClient();
    const { data: profile } = await svc.from('profiles').select('email').eq('id', userContext.userId).maybeSingle();
    const email = profile?.email ?? null;
    if (!(await isAdmin(userContext.userId, email))) {
      return new Response(JSON.stringify({ error: 'forbidden' }), { status: 403 });
    }
    try {
      const [circ, feat, shop, funnel] = await Promise.all([
        svc.from('v_merits_in_circulation').select('*'),
        svc.from('v_feature_consumption_daily').select('*'),
        svc.from('v_shop_purchases_daily').select('*'),
        svc.from('v_conversion_funnel').select('*'),
      ]);
      if (circ.error || feat.error || shop.error || funnel.error) {
        log.error('view-read-error', { circ: circ.error?.message, feat: feat.error?.message, shop: shop.error?.message, funnel: funnel.error?.message });
        return new Response(JSON.stringify({ error: 'metrics-unavailable' }), { status: 500 });
      }
      return new Response(JSON.stringify({
        merits_in_circulation: circ.data ?? [],
        feature_consumption:   feat.data ?? [],
        shop_purchases:        shop.data ?? [],
        conversion_funnel:     funnel.data ?? [],
      }), { status: 200, headers: { 'Content-Type': 'application/json' } });
    } catch (e) {
      log.error('admin-metrics-exception', { message: (e as Error).message });
      return new Response(JSON.stringify({ error: 'metrics-unavailable' }), { status: 500 });
    }
  },
});
```

import { handleRequest } from '../_shared/handler.ts';
import { getServiceClient } from '../_shared/supabaseClient.ts';
import { createLogger } from '../_shared/logger.ts';

const log = createLogger('check-merit-alerts');
const AI_FAILURE_THRESHOLD = 0.05;   // D-16, ROADMAP SC-6
const SHOP_FAILURE_THRESHOLD = 0.02; // D-16, ROADMAP SC-6
const DEBOUNCE_MINUTES = 60;

type AlertKey = 'negative_balance' | 'ai_failure_rate_high' | 'shop_failure_rate_high';

async function tryFireAlert(svc: ReturnType<typeof getServiceClient>, key: AlertKey, payload: Record<string, unknown>): Promise<'fired' | 'debounced' | 'db-error'> {
  const { data: prior, error: selErr } = await svc.from('alert_state').select('last_fired_at').eq('alert_key', key).maybeSingle();
  if (selErr) {
    log.warn('alert-state-select-error', { key, error: selErr.message });
    return 'db-error';
  }
  if (prior?.last_fired_at) {
    const ageMs = Date.now() - new Date(prior.last_fired_at).getTime();
    if (ageMs < DEBOUNCE_MINUTES * 60_000) return 'debounced';
  }
  // Upsert FIRST (Pitfall 7): debounce state must persist even if email later fails
  const { error: upErr } = await svc.from('alert_state').upsert({
    alert_key: key,
    last_fired_at: new Date().toISOString(),
    last_value: payload,
  });
  if (upErr) {
    log.warn('alert-state-upsert-error', { key, error: upErr.message });
    return 'db-error';
  }
  return 'fired';
}

async function dispatchEmail(subject: string, body: string): Promise<boolean> {
  // Forward through send-strivo-email edge fn (avoids a second Resend integration)
  const url = `${Deno.env.get('SUPABASE_URL')}/functions/v1/send-strivo-email`;
  const admins = (Deno.env.get('ADMIN_EMAILS') ?? '').split(',').map(s => s.trim()).filter(Boolean);
  if (admins.length === 0) { log.warn('no-admin-emails-configured'); return false; }
  const res = await fetch(url, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${Deno.env.get('INTERNAL_JOB_SECRET') ?? ''}`,
    },
    body: JSON.stringify({ to: admins[0], subject, html: body }),
  });
  return res.ok;
}

export default handleRequest({
  authMode: 'internal',
  handler: async () => {
    const svc = getServiceClient();
    let evaluated = 0;
    let dispatched = 0;
    let emailErrors = 0;

    // Evaluator 1: negative balance (SELECT from user_merits WHERE balance < 0)
    evaluated++;
    const { data: neg, error: negErr } = await svc.from('user_merits').select('user_id,balance').lt('balance', 0).limit(10);
    if (!negErr && neg && neg.length > 0) {
      const state = await tryFireAlert(svc, 'negative_balance', { count: neg.length, sample: neg });
      if (state === 'fired') {
        const ok = await dispatchEmail('[Strivo] Negative merit balance detected', `Users with negative balance: ${neg.length}`);
        if (ok) dispatched++; else emailErrors++;
      }
    }

    // Evaluator 2: AI failure rate
    evaluated++;
    const { data: aiRate, error: aiErr } = await svc.rpc('get_ai_failure_rate_1h');
    if (!aiErr && typeof aiRate === 'number' && aiRate >= AI_FAILURE_THRESHOLD) {
      const state = await tryFireAlert(svc, 'ai_failure_rate_high', { rate: aiRate, threshold: AI_FAILURE_THRESHOLD });
      if (state === 'fired') {
        const ok = await dispatchEmail('[Strivo] AI failure rate elevated', `1h failure rate: ${(aiRate*100).toFixed(1)}%`);
        if (ok) dispatched++; else emailErrors++;
      }
    }

    // Evaluator 3: Shop failure rate
    evaluated++;
    const { data: shopRate, error: shopErr } = await svc.rpc('get_shop_failure_rate_1h');
    if (!shopErr && typeof shopRate === 'number' && shopRate >= SHOP_FAILURE_THRESHOLD) {
      const state = await tryFireAlert(svc, 'shop_failure_rate_high', { rate: shopRate, threshold: SHOP_FAILURE_THRESHOLD });
      if (state === 'fired') {
        const ok = await dispatchEmail('[Strivo] Shop failure rate elevated', `1h failure rate: ${(shopRate*100).toFixed(1)}%`);
        if (ok) dispatched++; else emailErrors++;
      }
    }

    return new Response(JSON.stringify({ evaluated, dispatched, email_errors: emailErrors }), {
      status: 200,
      headers: { 'Content-Type': 'application/json' },
    });
  },
});
