    ```tsx
    import { motion, useReducedMotion } from 'framer-motion';
    import { Lock } from 'lucide-react';
    import { useNavigate } from 'react-router-dom';
    import { Button } from '@/components/ui/button';
    import { Badge } from '@/components/ui/badge';

    export interface LockedSectionProps {
      title: string;
      teaser: string;
      requiredTier: 'pro' | 'elite';

      teaser,
      requiredTier,
      ctaHref = '/pricing',
      className = '',

      const reduceMotion = useReducedMotion();
      const ctaLabel = requiredTier === 'elite' ? 'Upgrade to Elite' : 'Upgrade to Pro';
      const badgeText = requiredTier === 'elite' ? 'ELITE EXCLUSIVE' : 'PRO EXCLUSIVE';

      return (
        <motion.section
          data-testid="locked-section"

          initial={{ opacity: 0 }}

          <h3 className="text-xl font-semibold leading-tight mb-4">{title}</h3>

          <div className="relative">
            <div
              data-testid="locked-section-blur"
              aria-hidden="true"
              className="blur-[6px] opacity-40 select-none pointer-events-none space-y-2 py-4"
            >
              <div className="h-3 rounded bg-muted/40 w-[92%]" />
              <div className="h-3 rounded bg-muted/40 w-[78%]" />
              <div className="h-3 rounded bg-muted/40 w-[85%]" />
              <div className="h-3 rounded bg-muted/40 w-[66%]" />
              <div className="h-3 rounded bg-muted/40 w-[90%]" />

              </motion.div>
            </div>
          </div>

          <p
            data-testid="locked-section-teaser"
            className="text-base font-normal leading-normal text-muted-foreground mt-4"
          >
            {teaser}
          </p>

          <Button
            data-testid="locked-section-cta"
            className="mt-4"
            onClick={() => navigate(ctaHref)}
          >
            {ctaLabel}
          </Button>
        </motion.section>
      );
    }
    ```

    ```tsx
    import { Lock } from 'lucide-react';
    import { useNavigate } from 'react-router-dom';
    import { Button } from '@/components/ui/button';

    export interface LockedListItemProps {
      index: number;
      total: number;
      label: string;
      requiredTier: 'pro' | 'elite';
      teaser?: string;
      ctaHref?: string;
      className?: string;
    }

    export function LockedListItem({
      index,
      total,
      label,
      requiredTier,
      teaser,
      ctaHref = '/pricing',
      className = '',
    }: LockedListItemProps) {
      const navigate = useNavigate();
      const ctaLabel = requiredTier === 'elite' ? 'Upgrade to Elite' : 'Upgrade to Pro';
      // EN-DASH is U+2013, EM-DASH is U+2014. Spec says em-dash: '—'.

      return (
        <div
          data-testid="locked-list-item"
          data-index={index}
          className={`rounded-lg border border-border bg-card p-4 flex items-start gap-3 min-h-[var(--locked-item-height,4rem)] ${className}`}
        >

          <div className="flex-1">
            <p className="text-sm font-semibold text-muted-foreground">{lockLabel}</p>
            {teaser ? <p className="text-sm text-muted-foreground mt-1">{teaser}</p> : null}
            <Button
              variant="link"
              size="sm"
    ```
