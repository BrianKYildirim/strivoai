       Deno.test({
         name: 'TEST-10 resume Free token count < 50% of Elite (live, snapshot)',
         ignore: skip,
         async fn() {
           const freePrompt = loadPromptVariant('resume', 'free', {...params, tier:'free'});
           const elitePrompt = loadPromptVariant('resume', 'elite', {...params, tier:'elite', isElite:true});
           const freeResp = await invokeAnthropicText({ system: freePrompt, user: resumeText, maxTokens: 6000, model: getModel('free'), timeoutMs: getTimeout('free') });
           const eliteResp = await invokeAnthropicText({ system: elitePrompt, user: resumeText, maxTokens: 20000, model: getModel('elite', 'resume'), timeoutMs: getTimeout('elite') });

           const freeTokens = freeResp.usage?.output_tokens ?? estimateTokens(freeResp.text);
           const eliteTokens = eliteResp.usage?.output_tokens ?? estimateTokens(eliteResp.text);

           // Write snapshot
           const fixture = await readFixture();
           fixture['resume'] = {

             ratio: freeTokens / eliteTokens,
             timestamp: new Date().toISOString(),

         },
       });
