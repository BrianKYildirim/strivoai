/**
 * Phase 10 — Merit Ledger RLS denial tests (MER-04).
 *
 * These tests MUST run against a real Postgres (local `supabase start` or a
 * staging branch) because RLS cannot be faked. This is a deliberate departure
 * from the repo convention of colocated tests + fake supabase clients — the
 * test is not tied to a single edge function, so it lives in _tests/.
 *
 * Prerequisites:
 *   1. `supabase start` running locally OR SUPABASE_URL + SUPABASE_ANON_KEY
 *      pointing at a staging branch with the Phase 10 migration applied.
 *   2. TEST_USER_JWT exported — a valid JWT for a signed-in test user in that DB.
 *
 * Run with:
 *   deno test supabase/functions/_tests/merit_rls_test.ts --no-check --allow-env --allow-net
 *
 * Will skip with a clear message if prerequisites are missing so CI does not
 * red out when the migration is not yet deployed to the target environment.
 */
import { assertEquals, assert } from "https://deno.land/std@0.224.0/assert/mod.ts";
import { createClient } from "https://esm.sh/@supabase/supabase-js@2";

const SUPABASE_URL = Deno.env.get("SUPABASE_URL") ?? "";
const SUPABASE_ANON_KEY = Deno.env.get("SUPABASE_ANON_KEY") ?? "";
const TEST_USER_JWT = Deno.env.get("TEST_USER_JWT") ?? "";

const hasPrereqs = SUPABASE_URL && SUPABASE_ANON_KEY && TEST_USER_JWT;

function client() {
  return createClient(SUPABASE_URL, SUPABASE_ANON_KEY, {
    global: { headers: { Authorization: `Bearer ${TEST_USER_JWT}` } },
  });
}

function rlsDenied(error: { code?: string; message?: string } | null): boolean {
  if (!error) return false;
  // Postgres error code 42501 = insufficient_privilege (RLS denial surface)
  // PostgREST may also surface "new row violates row-level security policy"
  return (
    error.code === "42501" ||
    /row-level security/i.test(error.message ?? "") ||
    /permission denied/i.test(error.message ?? "")
  );
}

Deno.test("MER-04: anon client INSERT into user_merits blocked by RLS", async () => {
  if (!hasPrereqs) {
    console.warn("[SKIP] SUPABASE_URL / SUPABASE_ANON_KEY / TEST_USER_JWT not set");
    return;
  }
  const supabase = client();
  const { data, error } = await supabase.from("user_merits").insert({
    user_id: "00000000-0000-0000-0000-000000000000",
    subscription_balance: 999,
    shop_balance: 0,
    lifetime_granted: 999,
  }).select();

  assert(error !== null, `Expected RLS denial, got data: ${JSON.stringify(data)}`);
  assert(
    rlsDenied(error),
    `Expected RLS denial error, got: ${error?.code} ${error?.message}`,
  );
});

Deno.test("MER-04: anon client UPDATE to user_merits blocked by RLS", async () => {
  if (!hasPrereqs) {
    console.warn("[SKIP] SUPABASE_URL / SUPABASE_ANON_KEY / TEST_USER_JWT not set");
    return;

    .from("user_merits")
    .update({ subscription_balance: 9999 })

```powershell
<#
.SYNOPSIS
  Phase 10 / MIG-04 — capture read-only snapshot of legacy credit-ledger tables.
  Run BEFORE and AFTER applying the Phase 10 migration. Diff the two outputs to
  prove that usage_tracking / credit_usage / ai_usage are unmodified.

.DESCRIPTION
  For each legacy table, writes:
    - row count
    - md5 hash of a deterministic string_agg of all rows ordered by id
  into the output file. If a legacy table does not exist in the target DB,
  emits a MISSING marker (which is itself information — diff must match).

.PARAMETER DatabaseUrl
  Postgres connection string. Defaults to `$env:SUPABASE_DB_URL`.

.PARAMETER OutputFile
  Path to write the snapshot to. Defaults to `.snapshot-legacy.txt` at repo root.

.EXAMPLE
  .\scripts\snapshot-legacy-tables.ps1 -OutputFile .snapshot-legacy-pre.txt
  # ... run migration ...
  .\scripts\snapshot-legacy-tables.ps1 -OutputFile .snapshot-legacy-post.txt
  # Then:
  diff .snapshot-legacy-pre.txt .snapshot-legacy-post.txt
#>

param(
  [string]$DatabaseUrl = $env:SUPABASE_DB_URL,
  [string]$OutputFile = ".snapshot-legacy.txt"
)

$ErrorActionPreference = "Stop"

$repoRoot = Split-Path -Parent $PSScriptRoot
Set-Location $repoRoot

if ([string]::IsNullOrWhiteSpace($DatabaseUrl)) {
  Write-Error "DatabaseUrl not set. Pass -DatabaseUrl or set `$env:SUPABASE_DB_URL."
  exit 1
}

# Legacy ledger tables that MUST remain untouched by Phase 10 (MIG-04).
$legacyTables = @("usage_tracking", "credit_usage", "ai_usage")

$lines = @()
$lines += "# Phase 10 legacy ledger snapshot"
$lines += "# Generated: $(Get-Date -Format 'o')"
$lines += "# DatabaseUrl: (redacted)"
$lines += ""

foreach ($table in $legacyTables) {
  Write-Host "Snapshotting $table ..."

  # Exists check
  $existsSql = "SELECT to_regclass('public.$table') IS NOT NULL AS present;"
  $existsOut = & psql $DatabaseUrl -Atc $existsSql 2>&1
  if ($LASTEXITCODE -ne 0) {
    Write-Error "psql failed for exists-check on $table : $existsOut"
    exit 2
  }

  if ($existsOut.Trim() -ne "t") {
    $lines += "table=$table status=MISSING"
    continue
  }

  # Row count
  $countSql = "SELECT COUNT(*)::text FROM public.$table;"
  $countOut = & psql $DatabaseUrl -Atc $countSql
  if ($LASTEXITCODE -ne 0) { Write-Error "psql count failed for $table"; exit 3 }

  # Deterministic content hash: md5 of all columns concatenated per row, ordered by id
  # to_jsonb(t.*)::text is stable across psql versions and covers every column.
  $hashSql = @"
SELECT COALESCE(md5(string_agg(to_jsonb(t.*)::text, '|' ORDER BY t.id)), 'EMPTY')
FROM public.$table t;
"@
  $hashOut = & psql $DatabaseUrl -Atc $hashSql
  if ($LASTEXITCODE -ne 0) { Write-Error "psql hash failed for $table"; exit 4 }

  $lines += "table=$table status=PRESENT count=$($countOut.Trim()) md5=$($hashOut.Trim())"
}

$outputPath = Join-Path $repoRoot $OutputFile
$lines | Set-Content -Path $outputPath -Encoding UTF8

Write-Host "`nSnapshot written to: $outputPath"
Write-Host "Contents:"
Get-Content $outputPath | Write-Host
```

Notes:
- Script uses `psql` which ships with the Supabase CLI install — no additional dependency.
- Output is deterministic: same DB state → same file → `diff` returns empty.
- `to_jsonb(t.*)::text` captures every column including columns Phase 10 might mistakenly add (the whole point of MIG-04).
- If a legacy table doesn't exist in some environment, the MISSING marker is recorded so absence is still a stable signal.
