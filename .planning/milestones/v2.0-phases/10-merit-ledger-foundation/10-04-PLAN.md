**SC-1: Schema column shape assertions.** For each of the 3 new tables, query `information_schema.columns` and assert the expected (column_name, data_type, is_nullable) tuples are present.

```bash
psql "$SUPABASE_DB_URL" <<'SQL'
DO $$
DECLARE
  expected_cols text[];
  actual_cols   text[];
BEGIN
  -- user_merits expected set
  SELECT array_agg(c ORDER BY c) INTO expected_cols FROM (VALUES
    ('id|uuid|NO'),
    ('user_id|uuid|NO'),
    ('subscription_balance|integer|NO'),
    ('shop_balance|integer|NO'),
    ('lifetime_granted|integer|NO'),
    ('subscription_reset_date|timestamp with time zone|YES'),
    ('free_linkedin_scan_used|boolean|NO'),
    ('created_at|timestamp with time zone|NO'),
    ('updated_at|timestamp with time zone|NO')
  ) AS t(c);

  SELECT array_agg(column_name || '|' || data_type || '|' || is_nullable ORDER BY column_name || '|' || data_type || '|' || is_nullable)
    INTO actual_cols
    FROM information_schema.columns
    WHERE table_schema = 'public' AND table_name = 'user_merits';

  IF actual_cols IS DISTINCT FROM expected_cols THEN
    RAISE EXCEPTION 'SC-1 FAIL user_merits: expected % got %', expected_cols, actual_cols;
  END IF;

  -- merit_transactions expected set
  SELECT array_agg(c ORDER BY c) INTO expected_cols FROM (VALUES
    ('id|uuid|NO'),

    ('amount|integer|NO'),
    ('balance_type|text|NO'),
    ('source|text|NO'),
    ('feature|text|YES'),
    ('metadata|jsonb|NO'),
    ('created_at|timestamp with time zone|NO')
  ) AS t(c);

  SELECT array_agg(column_name || '|' || data_type || '|' || is_nullable ORDER BY column_name || '|' || data_type || '|' || is_nullable)
    INTO actual_cols
    FROM information_schema.columns
    WHERE table_schema = 'public' AND table_name = 'merit_transactions';

  IF actual_cols IS DISTINCT FROM expected_cols THEN
    RAISE EXCEPTION 'SC-1 FAIL merit_transactions: expected % got %', expected_cols, actual_cols;
  END IF;

  -- merit_shop_purchases expected set
  SELECT array_agg(c ORDER BY c) INTO expected_cols FROM (VALUES
    ('id|uuid|NO'),
    ('user_id|uuid|NO'),
    ('pack_size|integer|NO'),
    ('price_cents|integer|NO'),
    ('stripe_payment_intent_id|text|NO'),
    ('stripe_receipt_url|text|YES'),
    ('status|text|NO'),
    ('created_at|timestamp with time zone|NO')
  ) AS t(c);

  SELECT array_agg(column_name || '|' || data_type || '|' || is_nullable ORDER BY column_name || '|' || data_type || '|' || is_nullable)
    INTO actual_cols
    FROM information_schema.columns

    RAISE EXCEPTION 'SC-1 FAIL merit_shop_purchases: expected % got %', expected_cols, actual_cols;
```

**SC-3: CHECK constraint behavior.** Attempt to UPDATE a user_merits row to negative balance via service role (which bypasses RLS) — CHECK must still fire. Pick any existing row (the backfill guarantees at least one profile has a user_merits row if any profiles exist).

```bash
psql "$SUPABASE_DB_URL" <<'SQL'
DO $$
DECLARE
  any_user_id uuid;
  err_caught  boolean := false;
BEGIN
  SELECT user_id INTO any_user_id FROM public.user_merits LIMIT 1;
  IF any_user_id IS NULL THEN
    RAISE NOTICE 'SC-3 SKIP: no user_merits rows to test against (empty profiles table)';
    RETURN;
  END IF;

  BEGIN
    UPDATE public.user_merits SET subscription_balance = -1 WHERE user_id = any_user_id;
  EXCEPTION WHEN check_violation THEN
    err_caught := true;
  END;
  IF NOT err_caught THEN
    RAISE EXCEPTION 'SC-3 FAIL subscription_balance: UPDATE to -1 should have raised check_violation';
  END IF;

  err_caught := false;
  BEGIN
    UPDATE public.user_merits SET shop_balance = -1 WHERE user_id = any_user_id;
  EXCEPTION WHEN check_violation THEN
    err_caught := true;
  END;
  IF NOT err_caught THEN
    RAISE EXCEPTION 'SC-3 FAIL shop_balance: UPDATE to -1 should have raised check_violation';
  END IF;

  RAISE NOTICE 'SC-3 PASS: CHECK constraints block negative balances';
END $$;
SQL
```

```bash
psql "$SUPABASE_DB_URL" <<'SQL'
DO $$
DECLARE
  profile_count integer;
  merit_count   integer;
  correct_shape integer;
  grant_count   integer;
BEGIN
  SELECT COUNT(*) INTO profile_count FROM public.profiles;
  SELECT COUNT(*) INTO merit_count   FROM public.user_merits;
  SELECT COUNT(*) INTO correct_shape
    FROM public.user_merits
    WHERE subscription_balance = 5
      AND shop_balance = 0
      AND lifetime_granted = 5;
  SELECT COUNT(*) INTO grant_count
    FROM public.merit_transactions
    WHERE source IN ('phase-10-backfill','signup-grant');

  RAISE NOTICE 'profiles=% user_merits=% correct_shape=% grant_rows=%',
    profile_count, merit_count, correct_shape, grant_count;

  IF profile_count <> merit_count THEN
    RAISE EXCEPTION 'SC-4 FAIL: profile_count(%) != user_merits count(%)', profile_count, merit_count;
  END IF;

  IF profile_count <> correct_shape THEN
    RAISE EXCEPTION 'SC-4 FAIL: % profiles but only % user_merits with (sub=5, shop=0, lifetime=5)', profile_count, correct_shape;
  END IF;

  IF grant_count < profile_count THEN
    RAISE EXCEPTION 'SC-4 FAIL: % profiles but only % grant rows', profile_count, grant_count;
  END IF;

  RAISE NOTICE 'SC-4 PASS';
END $$;
```
