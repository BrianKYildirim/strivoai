```typescript
          // CC-04 gap closure: persist result to dashboard (mirrors legacy path lines 1153-1163)
          // Non-blocking: fire insert without awaiting so the response is not delayed
          const analysisResultSnapshot = JSON.parse(JSON.stringify(analysisResult));
          supabase
            .from('analysis_results')
            .insert({
              user_id: userContext.userId,
              analysis_type: 'career_compass',
              primary_score: analysisResultSnapshot.careerReadinessScore || analysisResultSnapshot.readinessScore || 0,
              key_insights: analysisResultSnapshot.whatYouHave || [],
              raw_response: analysisResultSnapshot,
            })
            .select('id')
            .single()
            .then(({ data, error }) => {
              if (error) {
                console.error('[analyze-career] Merit path analysis_results insert failed:', error.message);
              } else {
                console.log('[analyze-career] Merit path analysis_results saved, id:', data?.id);
                // Fire-and-forget completion email (same pattern as legacy path lines 1178-1189)
                if (data?.id) {
                  const APP_URL = Deno.env.get('APP_URL') || 'https://strivoai.net';
                  supabase
                    .from('profiles')
                    .select('email, full_name')
                    .eq('id', userContext.userId)
                    .single()
                    .then(({ data: profile }) => {
                      if (profile?.email) {
                        fetch(`${getSupabaseUrl()}/functions/v1/send-feature-email`, {
                          method: 'POST',
                          headers: {
                            'Content-Type': 'application/json',
                            'Authorization': `Bearer ${getServiceRoleKey()}`,
                            'x-internal-job-secret': Deno.env.get('INTERNAL_JOB_SECRET') || '',
                          },
                          body: JSON.stringify({
                            to: profile.email,
                            userName: profile.full_name || undefined,
                            featureType: 'career',
                            score: analysisResultSnapshot.careerReadinessScore || analysisResultSnapshot.readinessScore,
                            resultsUrl: `${APP_URL}/career-compass?id=${data.id}`,
                            summaryLine: analysisResultSnapshot.careerDNA?.slice(0, 200) || undefined,
                            recordId: data.id,
                            recordTable: 'analysis_results',
                          }),
                        }).then(res => {
                          if (res.ok) console.log('[analyze-career] Merit path completion email triggered');
                          else console.error('[analyze-career] Merit path email trigger failed:', res.status);
                        }).catch(err => console.error('[analyze-career] Merit path email trigger error:', err));
                      }
                    })
                    .catch(err => console.error('[analyze-career] Merit path profile lookup failed:', err));
                }
              }
            })
            .catch((err: Error) => {
              console.error('[analyze-career] Merit path analysis_results insert exception:', err.message);
            });
```

Key implementation notes:
- `JSON.parse(JSON.stringify(analysisResult))` creates a deep copy so the async insert captures the shaped data at this point in time, preventing any potential mutation issues if the reference is reused elsewhere
- The entire insert chain is fire-and-forget (no `await`) so it does not delay the Response
- The email sending is nested inside the insert success handler, matching the legacy path pattern where the email is sent only after a successful insert (since the email needs the `analysisRecord.id` for the results URL)
- Uses bracket-prefix logging pattern per project conventions (`[analyze-career]`)
