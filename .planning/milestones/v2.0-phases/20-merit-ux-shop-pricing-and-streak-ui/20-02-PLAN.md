Create this hook as a React Query `useQuery` wrapper over the `get_merit_balance` Supabase RPC:

```typescript
import { useQuery, useQueryClient } from '@tanstack/react-query';
import { supabase } from '@/lib/supabaseClient';
import { useEffect, useState } from 'react';

const LOW_BALANCE_THRESHOLD = 4; // D-05: 2x cheapest feature cost (2 Merits)

export interface MeritBalance {
  total: number;
  subscription: number;
  shop: number;
  resetDate: string | null;
  isLow: boolean;
  isZero: boolean;
  isLoading: boolean;
  canAfford: (cost: number) => boolean;
}

export function useMerits(): MeritBalance {
  const [userId, setUserId] = useState<string | null>(null);

  useEffect(() => {
    supabase.auth.getUser().then(({ data: { user } }) => {
      setUserId(user?.id ?? null);
    });
    const { data: { subscription } } = supabase.auth.onAuthStateChange((_event, session) => {
      setUserId(session?.user?.id ?? null);
    });
    return () => subscription.unsubscribe();
  }, []);

  const { data, isLoading } = useQuery({
    queryKey: ['merit-balance', userId],
    queryFn: async () => {
      const { data, error } = await supabase.rpc('get_merit_balance', {
        p_user_id: userId!,
      });
      if (error) throw error;
      return data;
    },
    enabled: !!userId,
    staleTime: 5_000,
    refetchOnWindowFocus: true,
  });

  const total = (data as any)?.total ?? 0;
  const subscription = (data as any)?.subscription_balance ?? 0;
  const shop = (data as any)?.shop_balance ?? 0;
  const resetDate = (data as any)?.next_reset_date ?? null;

  return {
    total,
    subscription,
    shop,
    resetDate,
    isLow: total > 0 && total < LOW_BALANCE_THRESHOLD,
    isZero: total <= 0,
    isLoading: isLoading || userId === null,
    canAfford: (cost: number) => total >= cost,
  };
}
```

**File 2: `src/hooks/useFeatureCosts.ts`** (per D-04)

```typescript
import { useQuery } from '@tanstack/react-query';
import { supabase } from '@/lib/supabaseClient';

interface FeatureCost {
  feature: string;
  tier: string;
  cost: number;
}

export function useFeatureCosts() {
  return useQuery({
    queryKey: ['feature-costs'],
    queryFn: async () => {
      const { data, error } = await supabase
        .from('merit_feature_costs')
        .select('feature, tier, cost');
      if (error) throw error;
      return (data ?? []) as FeatureCost[];
    },
    staleTime: 60_000, // costs rarely change; 1-minute cache
  });
}

export function getCostForFeature(
  costs: FeatureCost[] | undefined,
  feature: string,
  tier: string
): number | null {
  if (!costs) return null;
  const entry = costs.find((c) => c.feature === feature && c.tier === tier);
  return entry?.cost ?? null;
}
```

**File 3: `src/hooks/useStripeCheckout.ts`** (per D-10, D-16)

```typescript
import { useCallback, useState } from 'react';
import { supabase } from '@/lib/supabaseClient';
import { toast } from 'sonner';

interface StripeCheckoutParams {
  mode: 'subscription' | 'payment';
  tier?: 'pro' | 'elite';
  interval?: 'monthly' | 'annual';
  pack_size?: number;
}

interface UseStripeCheckoutReturn {
  isLoading: boolean;
  fetchClientSecret: (params: StripeCheckoutParams) => Promise<string | null>;
}

export function useStripeCheckout(): UseStripeCheckoutReturn {
  const [isLoading, setIsLoading] = useState(false);

  const fetchClientSecret = useCallback(async (params: StripeCheckoutParams): Promise<string | null> => {
    setIsLoading(true);
    try {
      const { data: { user } } = await supabase.auth.getUser();
      if (!user) {
        toast.error('Please sign in first');
        setIsLoading(false);
        return null;
      }

      const { data, error } = await supabase.functions.invoke('create-checkout-session', {
        body: { ...params, ui_mode: 'embedded' },
      });

      if (error) {
        toast.error('Could not start checkout. Please try again.');
        console.error('[useStripeCheckout] invoke error', error);
        setIsLoading(false);
        return null;
      }

      setIsLoading(false);
      return data?.clientSecret ?? null;
    } catch (err) {
      console.error('[useStripeCheckout] unexpected error', err);
      toast.error('Payment could not be completed. Please try again.');
      setIsLoading(false);
      return null;
    }
  }, []);

  return { isLoading, fetchClientSecret };
}
```
