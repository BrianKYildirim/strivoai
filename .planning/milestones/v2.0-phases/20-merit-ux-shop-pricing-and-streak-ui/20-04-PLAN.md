```typescript
import { useCallback, useState } from 'react';
import { loadStripe } from '@stripe/stripe-js';
import {
  EmbeddedCheckoutProvider,
  EmbeddedCheckout,
} from '@stripe/react-stripe-js';
import {
  Dialog,
  DialogContent,
  DialogHeader,
  DialogTitle,
} from '@/components/ui/dialog';
import { Loader2 } from 'lucide-react';

const stripePromise = loadStripe(import.meta.env.VITE_STRIPE_PUBLISHABLE_KEY || '');

interface StripeCheckoutWrapperProps {
  open: boolean;
  onOpenChange: (open: boolean) => void;
  clientSecret: string | null;
  title: string;
}

export function StripeCheckoutWrapper({
  open,
  onOpenChange,
  clientSecret,
  title,
}: StripeCheckoutWrapperProps) {
  if (!clientSecret) return null;

  return (
    <Dialog open={open} onOpenChange={onOpenChange}>
      <DialogContent className="max-w-[560px] max-h-[90vh] overflow-y-auto">
        <DialogHeader>
          <DialogTitle>{title}</DialogTitle>
        </DialogHeader>
        <EmbeddedCheckoutProvider
          stripe={stripePromise}
          options={{ clientSecret }}
        >
          <EmbeddedCheckout />
        </EmbeddedCheckoutProvider>
      </DialogContent>
    </Dialog>
  );
}
```

Key points:
- `stripePromise` is created once at module level using `loadStripe()` with `VITE_STRIPE_PUBLISHABLE_KEY`
- The Dialog has `max-w-[560px]` per UI-SPEC
- `clientSecret` is passed as a prop (the parent component fetches it via `useStripeCheckout`)
- Loading state is handled by the parent before passing clientSecret
- EmbeddedCheckout handles its own internal loading/success states

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-20-09 | Tampering | Price display vs actual charge | accept | Prices displayed are static constants matching Stripe Price objects. Actual charge is determined server-side by the resolved Price ID. Display mismatch has no security impact. |
| T-20-10 | Information Disclosure | VITE_STRIPE_PUBLISHABLE_KEY | accept | Stripe publishable keys are designed to be public; they cannot be used for server-side operations |
