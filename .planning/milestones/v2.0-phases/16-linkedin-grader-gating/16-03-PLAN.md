/**
 * Defense-in-depth strip for Free-tier LinkedIn output (D-01, D-03).
 * Runs AFTER validateLinkedInGrade() -- the full shape has been validated.
 *
 * LIG-01 keeps: rawScore, scoreJustification, keyStrengths (full), nextBestAction,
 *               topFixes[0] only, sectionBreakdown with scores only (no feedback).
 * LIG-02 removes: profileInsights entirely, topFixes[1+], all feedback arrays.
 *
 * Exported for unit testing (strip.test.ts).
 */
export function stripLinkedInFreeTier(parsed: Record<string, unknown>): Record<string, unknown> {
  // LIG-02: remove profileInsights entirely
  delete parsed.profileInsights;

  // LIG-01: keep only first topFixes entry
  if (Array.isArray(parsed.topFixes)) {
    parsed.topFixes = parsed.topFixes.length > 0 ? [parsed.topFixes[0]] : [];
  }

  // LIG-02: strip feedback arrays from sectionBreakdown (keep name, score, maxScore, status)
  if (Array.isArray(parsed.sectionBreakdown)) {
    for (const section of parsed.sectionBreakdown) {
      if (section && typeof section === 'object') {
        delete (section as Record<string, unknown>).feedback;
      }
    }
  }

  return parsed;
}

   ```typescript
   /**
    * Executes the tier-aware AI call with retry and validation.
    * Used by both the free-scan path and the withMerits callback.
    */
   async function runTieredAiCall(
     tier: Tier,
     userId: string,
     truncatedContent: string,
     profileUrl: string,
     followUpContext: string | null,
     log: ReturnType<typeof createLogger>,
   ): Promise<{ parsed: Record<string, unknown>; response?: never } | { parsed?: never; response: Response }> {
     const systemPrompt = loadPromptVariant('linkedin', tier, {});
     const userMessage = buildUserMessage(truncatedContent, profileUrl || undefined, followUpContext);

     const aiParams = {
       model: getModel(tier, 'linkedin'),
       system: systemPrompt,
       messages: [{ role: 'user' as const, content: userMessage }],
       maxTokens: LINKEDIN_MAX_TOKENS[tier],
       temperature: 0.3,
       timeoutMs: getTimeout(tier),
       fixtureKey: 'grade_linkedin',
     };

     const ai = await invokeWithRetry(aiParams, log);
     if (!ai.ok) throw new MeritRefundable(ai.status, ai.error);

     let parsed: unknown;
     try { parsed = parseAiResponse(ai.text); } catch { parsed = null; }
     let validation = parsed ? validateLinkedInGrade(parsed) : { valid: false as const, error: 'JSON parse failed' };
     if (validation.correctedScore !== undefined) {
       (parsed as Record<string, unknown>).rawScore = validation.correctedScore;
       validation = validateLinkedInGrade(parsed);
     }

     if (!validation.valid) {
       log.warn('validation-failed-first-attempt', { error: validation.error, tier });
       const retryAi = await invokeWithRetry(aiParams, log);
       if (!retryAi.ok) throw new MeritRefundable(retryAi.status, retryAi.error);
       try { parsed = parseAiResponse(retryAi.text); } catch {
         throw new MeritRefundable(500, 'Failed to parse LinkedIn grade after retry');
       }
       validation = validateLinkedInGrade(parsed);
       if (validation.correctedScore !== undefined) {
         (parsed as Record<string, unknown>).rawScore = validation.correctedScore;
         validation = validateLinkedInGrade(parsed);
       }
       if (!validation.valid) throw new MeritRefundable(500, 'Validation failed after retry');
     }

     // D-01: Apply free-tier strip AFTER validation (defense-in-depth)
     if (tier === 'free') {
       stripLinkedInFreeTier(parsed as Record<string, unknown>);
     }

     return { parsed: parsed as Record<string, unknown> };
   }
   ```
