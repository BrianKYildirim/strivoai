    ```sql
    -- 2. SUB-11/SUB-12/NFR-08: cron sweep for monthly Merit refills (yearly subs + fallback for monthly).
    --    IMPORTANT: this cron.schedule block is intended to be applied per-environment via Supabase
    --    Studio (mirrors the keep-warm migration) because the URL + internal secret differ per project.
    --    Staging project ref:    cimjltzovkahkqecpwgi
    --    Production project ref: zjpjapgraoxdukdvpjes
    --    For `supabase db push`, the CREATE EXTENSION is a no-op if already enabled; the
    --    cron.schedule call uses an `ON CONFLICT DO NOTHING`-equivalent pattern via cron.unschedule-if-exists
    --    so the migration is re-runnable.
    CREATE EXTENSION IF NOT EXISTS pg_cron;
    CREATE EXTENSION IF NOT EXISTS pg_net;

    DO $cron$
    BEGIN
      IF EXISTS (SELECT 1 FROM cron.job WHERE jobname = 'reset-subscription-merits-sweep') THEN
        PERFORM cron.unschedule('reset-subscription-merits-sweep');
      END IF;

      PERFORM cron.schedule(
        'reset-subscription-merits-sweep',
        '*/15 * * * *',
        $job$
        SELECT net.http_post(
          url     := current_setting('app.functions_base_url', true) || '/reset-subscription-merits',
          headers := jsonb_build_object(
            'Content-Type', 'application/json',
            'x-internal-job-secret', current_setting('app.internal_job_secret', true)
          ),
          body    := '{"invocation_source": "pg_cron"}'::jsonb
        );
        $job$
      );
    END
    $cron$;
    ```

    Section 3 — MIG-03 backfill DO block with guard (mirrors Phase 10 D-15 `source = 'phase-12-backfill'` convention):
    ```sql
    -- 3. MIG-03: backfill existing active Pro/Elite subscribers. Runs once.
    --    Guards on Phase 11 RPC existence — fails loudly if migration ordering is wrong.
    DO $backfill$
    DECLARE
      r RECORD;
      credited_count integer := 0;
      rpc_exists boolean;
    BEGIN
      -- Guard: Phase 11 reset_subscription_merits must exist
      SELECT EXISTS (
        SELECT 1 FROM pg_proc p
          JOIN pg_namespace n ON n.oid = p.pronamespace
        WHERE n.nspname = 'public' AND p.proname = 'reset_subscription_merits'
      ) INTO rpc_exists;

      IF NOT rpc_exists THEN
        RAISE EXCEPTION 'Phase 11 RPC reset_subscription_merits not deployed — apply Phase 11 migration first';
      END IF;

      FOR r IN
        SELECT id AS user_id, subscription_tier
        FROM public.profiles
        WHERE subscription_tier IN ('pro','elite')
          AND subscription_status = 'active'
      LOOP
        BEGIN
          PERFORM public.reset_subscription_merits(
            p_user_id         := r.user_id,
            p_tier            := r.subscription_tier,
            p_next_reset_date := (now() + interval '1 month')
          );

          INSERT INTO public.merit_transactions(user_id, amount, balance_type, source, feature, metadata)
          VALUES (
            r.user_id,
            CASE r.subscription_tier WHEN 'pro' THEN 75 WHEN 'elite' THEN 150 ELSE 0 END,
            'subscription',
            'phase-12-backfill',
            NULL,
            jsonb_build_object('tier', r.subscription_tier, 'migration', '20260415000000')
          );

          credited_count := credited_count + 1;
        EXCEPTION WHEN OTHERS THEN
          RAISE WARNING 'phase-12-backfill skip user_id=%: %', r.user_id, SQLERRM;
        END;
      END LOOP;

      RAISE NOTICE 'phase-12-backfill complete: credited % active subscribers', credited_count;
    END
    $backfill$;
    ```
