    function mockGrantSubscriptionMerits(
      ledger: MockLedger, userId: string, tier: "pro" | "elite"
    ): Record<string, unknown> {
      // Mirrors public.grant_subscription_merits (20260415000000_merit_rpcs.sql §3).
      const allowance = tier === "pro" ? 75 : 150;
      ledger.subscription_balance = allowance; // Fresh allowance; no rollover per NFR-08.
      const txId = crypto.randomUUID();
      ledger.transactions.push({
        id: txId, user_id: userId, amount: allowance, balance_type: 'grant',
        source: 'subscription', feature: null, metadata: { tier },
        created_at: new Date().toISOString(),
      });
      return { success: true, new_subscription_balance: allowance, transaction_id: txId };
    }

    function mockGrantUpgradeDiff(
      ledger: MockLedger, userId: string, fromTier: string, toTier: string
    ): Record<string, unknown> {
      // Mirrors public.grant_upgrade_diff (20260415000000_merit_rpcs.sql §6).
      const from = fromTier === "pro" ? 75 : fromTier === "elite" ? 150 : 0;
      const to = toTier === "pro" ? 75 : toTier === "elite" ? 150 : 0;
      const diff = Math.max(0, to - from);
      ledger.subscription_balance += diff;
      const txId = crypto.randomUUID();
      ledger.transactions.push({
        id: txId, user_id: userId, amount: diff, balance_type: 'grant',
        source: 'upgrade-diff', feature: null,
        metadata: { from_tier: fromTier, to_tier: toTier },
        created_at: new Date().toISOString(),
      });
      return { success: true, diff_credited: diff, new_subscription_balance: ledger.subscription_balance, transaction_id: txId };
    }

    function tierGatedRecruitingLensOutput(tier: "free" | "pro" | "elite") {

    Deno.test("TEST-05: Free -> Pro -> Recruiting Lens -> Pro -> Elite -> Recruiting Lens (tier-gated)", async () => {

      const evCreated = makeSubscriptionCreatedEvent({ userId: USER, tier: "pro" });
      const body1 = JSON.stringify(evCreated);
      const sig1 = await signStripePayload(body1, TEST_SECRET);
      const req1 = new Request("https://example.com/stripe-webhook-handler", {
        method: "POST",
        headers: { "stripe-signature": sig1, "Content-Type": "application/json" },
        body: body1,

      const fakeStripe = buildFakeStripe(evCreated);

      assertEquals(resp1.status, 200);
      assertEquals(ledger.subscription_balance, 75);

      const r2 = mockCheckAndDeductMerits(ledger, USER, "recruiting_lens", "pro");
