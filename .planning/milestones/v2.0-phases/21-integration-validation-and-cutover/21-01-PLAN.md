    ```ts
    function mockCreditShopMerits(
      ledger: MockLedger, userId: string, packSize: number, stripePaymentId: string
    ): Record<string, unknown> {
      // Idempotency: if transactions[] already has a row with metadata.stripe_payment_id === stripePaymentId, return { success: false, error: 'duplicate_payment' }.
      const duplicate = ledger.transactions.find(t => t.metadata?.stripe_payment_id === stripePaymentId);
      if (duplicate) return { success: false, error: 'duplicate_payment', transaction_id: duplicate.id };
      // Validate pack size
      if (![3, 10, 25, 50].includes(packSize)) return { error: 'invalid_pack_size' };
      // Credit
      ledger.shop_balance += packSize;
      const txId = crypto.randomUUID();
      ledger.transactions.push({
        id: txId, user_id: userId, amount: packSize, balance_type: 'grant',
        source: 'shop', feature: null,
        metadata: { stripe_payment_id: stripePaymentId, pack_size: packSize },
        created_at: new Date().toISOString(),
      });
      return { success: true, transaction_id: txId, new_shop_balance: ledger.shop_balance };
    }
    ```

    ```ts
    Deno.test("TEST-04: Free signup -> 5 Merits -> Resume (3) -> Career Compass blocked -> shop 3-pack -> Career Compass succeeds", async () => {
      const USER = "00000000-0000-0000-0000-test04000001";
      const TEST_SECRET = "whsec_test_secret_abcdef1234567890";
      const ledger = createMockLedger(5, 0); // 5 lifetime Merits in subscription_balance per MER-10

      // Step 1: Resume Analysis succeeds (3 deducted)
      const r1 = mockCheckAndDeductMerits(ledger, USER, "resume_analysis", "free");
      assertEquals((r1 as any).success, true);
      assertEquals((r1 as any).remaining_balance, 2);
      assertEquals((r1 as any).deducted_from.subscription, 3);
      assertEquals((r1 as any).deducted_from.shop, 0);

      // Step 2: Career Compass blocked (need 3, have 2)
      const r2 = mockCheckAndDeductMerits(ledger, USER, "career_compass", "free");
      assertEquals((r2 as any).error, "insufficient_merits");
      assertEquals(ledger.subscription_balance + ledger.shop_balance, 2, "balance unchanged on block");

      // Step 3: Signed shop-3-pack webhook credits 3 to shop_balance
      const event = makeCheckoutSessionCompletedEvent({
        mode: "payment", packSize: 3,
        paymentIntentId: "pi_test04_3pack", userId: USER,
      });
      const body = JSON.stringify(event);
      const signature = await signStripePayload(body, TEST_SECRET);
      const req = new Request("https://example.com/stripe-webhook-handler", {
        method: "POST",
        headers: { "stripe-signature": signature, "Content-Type": "application/json" },
        body,
      });
      const fakeSupabase = buildMockServiceClient(ledger, USER);
      const fakeStripe = buildFakeStripe(event);
      const resp = await handleStripeWebhook(req, { supabase: fakeSupabase, stripe: fakeStripe });
      assertEquals(resp.status, 200);
      assertEquals(ledger.shop_balance, 3);
      assertEquals(ledger.subscription_balance + ledger.shop_balance, 5);

      // Step 4: Career Compass now succeeds; subscription-first ordering
      const r4 = mockCheckAndDeductMerits(ledger, USER, "career_compass", "free");
      assertEquals((r4 as any).success, true);
      assertEquals((r4 as any).remaining_balance, 2);
      assertEquals((r4 as any).deducted_from.subscription, 2);
      assertEquals((r4 as any).deducted_from.shop, 1);
    });

    Deno.test("TEST-04 idempotency: re-delivery of same Stripe event does not double-credit", async () => {
      const USER = "00000000-0000-0000-0000-test04000002";
      const TEST_SECRET = "whsec_test_secret_abcdef1234567890";

      assertEquals(ledger.shop_balance, 3, "second delivery must not re-credit");
    });
    ```
