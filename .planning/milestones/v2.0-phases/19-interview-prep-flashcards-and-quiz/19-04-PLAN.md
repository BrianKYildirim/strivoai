3. **handleGenerate function:**
   ```typescript
   const handleGenerate = async (jobDescription: string, jobTitle?: string, company?: string) => {
     if (!canAnalyze) { triggerPaywall(); return; }

     setIsAnalyzing(true);
     setPageState('generating');
     setErrorInfo(null);

     const result = await startAnalysis(async (_signal) => {
       const { data, error } = await invokeWithTimeout('interview-prep-analyze', {
         body: {
           jobDescription: jobDescription.trim(),
           jobTitle: jobTitle?.trim() || undefined,
           company: company?.trim() || undefined,
         },
       }, 120000);

       if (error) {
         if (data && typeof data === 'object' && 'error' in data) {
           const errCode = (data as any).error;
           if (errCode === 'no_resume') throw new Error((data as any).message);
           if (errCode === 'daily_cap_reached') throw new Error((data as any).message);
           if (errCode === 'pro_or_elite_required') throw new Error((data as any).message);
         }
         throw new Error(error.message || 'Generation failed. Please try again.');
       }
       return data;
     });

     setIsAnalyzing(false);

     if (result.success && result.data) {
       const data = result.data as any;
       if (data.mode === 'flashcard') {
         setFlashcards(data.flashcards || []);
         setPageState('flashcard-results');
       } else if (data.mode === 'quiz') {
         setQuizId(data.quiz_id);
         setQuizQuestions(data.questions || []);
         if (data.streak) setStreakInfo(data.streak);
         if (data.daily_usage) setDailyUsage(data.daily_usage);
         setPageState('quiz-active');
       }
       toast.success(data.mode === 'flashcard' ? 'Flashcards generated!' : 'Quiz ready!');
       refreshPaywall();
     } else if (result.error) {
       console.error('[InterviewPrep] error:', result.error);
       setErrorInfo({ type: 'generation', message: result.error });
       setPageState('error');
       toast.error(result.error);
     }
   };
   ```
