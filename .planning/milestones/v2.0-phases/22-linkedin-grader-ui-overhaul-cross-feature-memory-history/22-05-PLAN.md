```tsx
import { useSearchParams } from 'react-router-dom';
import { toast } from 'sonner';

const [searchParams] = useSearchParams();
const idFromUrl = searchParams.get('id');

useEffect(() => {
  if (!idFromUrl) return;
  // V5 (input validation) — validate UUID before DB call
  if (!/^[0-9a-f-]{36}$/i.test(idFromUrl)) {
    toast.error('Analysis not found');
    return;
  }
  let cancelled = false;
  (async () => {
    const { data, error } = await supabase
      .from('<feature_table>')
      .select('*')
      .eq('id', idFromUrl)
      // optional: .eq('analysis_type', '<type>') for analysis_results-backed pages
      .maybeSingle();
    if (cancelled) return;
    if (error || !data) {
      toast.error('Analysis not found');
      return;
    }
    // set result/analysis state + hide input view
    setResult(data); // variable names vary per page
    setShowInput(false);
  })();
  return () => { cancelled = true; };
}, [idFromUrl]);
```
