```sql
-- user_merits table schema
CREATE TABLE IF NOT EXISTS public.user_merits (
  id                       uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id                  uuid NOT NULL UNIQUE REFERENCES public.profiles(id) ON DELETE CASCADE,
  subscription_balance     integer NOT NULL DEFAULT 0 CHECK (subscription_balance >= 0),
  shop_balance             integer NOT NULL DEFAULT 0 CHECK (shop_balance >= 0),
  lifetime_granted         integer NOT NULL DEFAULT 0,
  subscription_reset_date  timestamptz NULL,
  free_linkedin_scan_used  boolean NOT NULL DEFAULT false,
  created_at               timestamptz NOT NULL DEFAULT now(),
  updated_at               timestamptz NOT NULL DEFAULT now()
);

-- merit_transactions table schema (audit trail)
CREATE TABLE IF NOT EXISTS public.merit_transactions (
  id            uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id       uuid NOT NULL REFERENCES public.profiles(id) ON DELETE CASCADE,
  amount        integer NOT NULL,
  balance_type  text NOT NULL CHECK (balance_type IN ('subscription','shop','grant','refund')),
  source        text NOT NULL,
  feature       text NULL,
  metadata      jsonb NOT NULL DEFAULT '{}'::jsonb,
  created_at    timestamptz NOT NULL DEFAULT now()
);
```

**1. `admin_set_user_merits(p_user_id UUID, p_merits INTEGER)`**
- SECURITY DEFINER, search_path = public, pg_temp (match Phase 11 RPC hardening pattern)
- Verify caller is admin via `public.has_role(auth.uid(), 'admin')` — raise exception if not
- Validate p_merits >= 0
- Retrieve the user's current subscription_balance from user_merits (SELECT ... FOR UPDATE to lock the row)
- If no user_merits row found, RAISE EXCEPTION 'User merit record not found'
- Compute delta: `p_merits - current_subscription_balance`
- UPDATE user_merits SET subscription_balance = p_merits, updated_at = now() WHERE user_id = p_user_id
- INSERT into merit_transactions: amount = delta, balance_type = 'grant', source = 'admin-adjustment', metadata = jsonb_build_object('admin_id', auth.uid(), 'previous_balance', old_balance, 'new_balance', p_merits)
- RETURN p_merits (the new balance)
- REVOKE ALL FROM public, GRANT EXECUTE TO authenticated, service_role (same pattern as Phase 11 RPCs)

**2. Updated `get_admin_usage_stats()`**
- DROP the existing function first (return type is changing)
- New return type: user_id uuid, email text, subscription_tier text, total_actions bigint, last_action_at timestamptz, subscription_balance integer, shop_balance integer, total_merits integer
